% evidence-anchor: theorem-6.1-signed-future-last-insertion
\subsection*{Theorem 6.1 (complete signed last-insertion transpose)}

For every fixed finite chart for which the displayed current terms lie in
$L^2$ and the trace pairings lie in $L^1$,
\[
\boxed{
\begin{aligned}

 =\sum_e\sum_{\ell<r<t_e}\bigg[&

  \langle d_rJ^{(e-1)},d_r\delta_eJ\rangle

 &-{1\over2}\mathbb E\int\delta_eB:\Delta\Gamma_r

\end{aligned}}                                                   \tag{6.3}

Equivalently, with
\[

   \langle Q_eJ^{(e-1)},Q_e\delta_eJ\rangle

and $\sum_{r>\ell}\mathfrak I_r=\sum_e\mathfrak I_e^{\rm LI}$.

\emph{Proof.}
