\[
 \boxed{

Hence neither a moving-endpoint secant nor a stationary subtraction may drop

\section*{3. Direct signed covariance-normal differentiation}

\[
 \boxed{
 D^2\cE_{\rm CN}[H,K]
 =\Rea\E_f\{\ip{\dot J_H}{\dot J_K}

nonnegative derivative Gram on the diagonal and one signed curvature

\section*{4. Analytic shortcuts tested and rejected}

\subsection*{4.1 Conditional Poincare does not give parameter semiconvexity}

For a standard Gaussian $\eta$ and
\[
                         J_N(z,\eta)=\cos(Nz)\eta,       \tag{4.1}
\]
the Gaussian Poincare inequality is an equality:
\[
 \Var(J_N)=\cos^2(Nz)=\E|D_\eta J_N|^2.                 \tag{4.2}
\]
Nevertheless
\[
                         {\dd^2\over\dd z^2}\Var(J_N)(0)
                         =-2N^2.                         \tag{4.3}
\]
Differentiating a Poincare upper bound therefore cannot supply a uniform
parameter Hessian bound.  This is an abstract diagnostic, not a production
counterexample.

\subsection*{4.2 Gaussian second score is exact but moment-expensive}

For deterministic or strict-past $u,v$ and sufficiently integrable $F$,
second Gaussian integration by parts gives
\[
 \E D_\xi^2F[u,v]
 =\E[(\ip{\xi}{u}\ip{\xi}{v}-\ip{u}{v})F].             \tag{4.4}
\]
The score has exact $L^2$ norm
\[
 \sqrt{\|u\|^2\|v\|^2+\ip{u}{v}^2}
 \leq\sqrt2\|u\|\|v\|,                               \tag{4.5}
\]
and $\sqrt2$ is sharp for parallel vectors.  The route removes explicit
feedback jets only after complete Gaussian integration, but it demands an
$L^2$ endpoint bound.  On the R-122 coherent-amplitude family the relevant
current-square endpoint reaches the eighth moment
$\E A_t^8=e^{8t^2}$.  No registered source/sextic graph budget controls that
separated moment uniformly.  The route is therefore parked unless a complete
signed endpoint supplies the missing cancellation before Cauchy--Schwarz.

\subsection*{4.3 Entropy does not control the second score}

Let $\gamma$ be the standard Gaussian law on $\mathbb R$ and
\[
 \rho_N(x)=1+\epsilon\sin(Nx),
 \qquad 0<\epsilon<1.                                   \tag{4.6}
\]
Then $\int\rho_N\dd\gamma=1$ and the elementary inequality
$(1+u)\log(1+u)\leq u+u^2$ gives
\[
 \operatorname{Ent}_\gamma(\rho_N)\leq{\epsilon^2\over2}.  \tag{4.7}
\]
The Fisher information instead obeys
\[
 I_\gamma(\rho_N)
 =\epsilon^2N^2\E_\gamma
   {\cos^2(NX)\over1+\epsilon\sin(NX)}
 \geq{\epsilon^2N^2(1+e^{-2N^2})\over2(1+\epsilon)}.   \tag{4.8}
\]
It diverges quadratically.  Normalized entropy and Doob orthogonality cannot
control the conditional Fisher or second-score terms; for a Gibbs density
those terms contain the force and Hessian being sought and the argument is
circular.

\subsection*{4.4 A rational second derivative is not separately floor-uniform}

For a homogeneous rational A1 row, scaling the floor $e>0$ at
$u=\sqrt e\,v$ gives
\[
 c_e(\sqrt e\,v)=\sqrt e\,c_1(v),
 \quad
 Dc_e(\sqrt e\,v)=Dc_1(v),
 \quad
 D^2c_e(\sqrt e\,v)=e^{-1/2}D^2c_1(v).                 \tag{4.9}
\]
An unweighted $D^2c$ supremum is therefore not floor-uniform.  The complete
products $cD^2c$ and $(Dc)^2$ are scale-neutral and must remain inside the
signed owner-complete Hessian.

\section*{7. Exact shifted-Douglas uniform-gap theorem}

Use the R-128 full-cross convention and put

\subsection*{Theorem 7.1 (shifted Douglas criterion)}

For $\mu\geq0$, define
\[
                         D_\mu=D-\mu I.                 \tag{7.3}
\]
Then
\[
 \boxed{\mathcal M\succeq\mu I}                        \tag{7.4}
\]
if and only if
\[
 \boxed{
 D_\mu\succeq0,
 \quad
 \Ran K^*\subseteq\Ran D_\mu^{1/2},
 \quad
 M_2-\mu I-X_\mu^*X_\mu\succeq0,}                    \tag{7.5}
\]
where $X_\mu$ is the Douglas reduced solution of
\[
                         D_\mu^{1/2}X_\mu=K^*.          \tag{7.6}
\]
A Moore--Penrose expression may be used only after closed range is proved.

\emph{Proof.}
Subtract $\mu I$ from the three-channel matrix and apply the bounded-operator
Douglas--Schur theorem to the lower-right block $D_\mu$. \hfill$\square$

\subsection*{Corollary 7.2 (norm-level three-channel gap)}

Assume
\[
 E\succeq eI,
 \quad F_0\succeq fI,
 \quad \|A_{\rm eff}\|\leq a,
 \quad D\succeq dI,
 \quad \|K\|\leq k.                                    \tag{7.7}
\]

 m_2={e+f-\sqrt{(e-f)^2+a^2}\over2}.                    \tag{7.8}

\[

 \qquad                 k^2<dm_2,                      \tag{7.9}

\[
 \boxed{
 \mathcal M\succeq\mu_3I,
 \qquad
 \mu_3={m_2+d-\sqrt{(m_2-d)^2+4k^2}\over2}>0.}         \tag{7.10}

This follows by first replacing $M_2$ by $m_2I$ and then taking the least
eigenvalue of the remaining scalar two-by-two matrix.

\[

 =\mu+{k^2\over d-\mu}.                                 \tag{7.11}

\[
 e>\tau_\mu,
 \qquad f>\tau_\mu,

At the R-128 zero-diagonal half-debt specialization,

The number (7.13) is not a general production margin: it assumes

In the scalar three-channel case the exact Schur determinant reads

\left(2\zeta-{c^2\over d}\right)
