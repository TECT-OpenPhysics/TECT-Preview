\[

\[

% evidence-anchor: theorem-3.1-complete-gram-row-heat-telescope
\subsection*{Theorem 3.1 (complete Gram-row heat martingale)}

For every admissible $H\in\cF_{r-1}$,
\[
 \boxed{
 \E\bigl[\mathcal W_{r,B}^{(J)}(H)\mid\cF_{r-1}\bigr]
 =\mathcal W_{r-1,B}^{(J)}(H).}                           \tag{3.1}
\]

\emph{Proof.}
Condition on $\cF_{r-1}$ and at one spatial point put
\[
 z=X_{r-1}+H,\qquad y_i=D_iX_{r-1}+D_iH.
\]
By (2.3), $g_r$ and $d_{r,i}$ are independent.  Hence
\[
\begin{aligned}
 &\E_{r-1}\left[
 \overline B_r^{(J)}(z+g_r):
 \bigl((y_i+d_{r,i})^{\otimes2}-\Gamma_{r,i}\bigr)
 \right]\\
 &\quad=
 \E_{g_r}\overline B_r^{(J)}(z+g_r):
 \bigl(y_i^{\otimes2}-\Gamma_{r-1,i}\bigr).
\end{aligned}                                             \tag{3.2}
\]
Heat-semigroup composition gives
\[
 \E_{g_r}\overline B_r^{(J)}(z+g_r)
 =P_{\Delta\Sigma_r}\overline B_r^{(J)}(z)
 =\overline B_{r-1}^{(J)}(z).                             \tag{3.3}
\]
Sum in $i$, integrate in $x$, and use finite-cutoff Fubini. \qed

Let
\[
 a_k\in\cF_{k-1},\qquad
 A^{(k)}=A^0+\sum_{j_0\leq s\leq k}a_s.                 \tag{3.4}
\]
Define
\[

 =\mathcal W_{k-1,B}^{(J)}(A^{(k)})
  -\mathcal W_{k-1,B}^{(J)}(A^{(k-1)})                   \tag{3.5}
\]
and
\[
 \mathfrak M_{k,B}
 =\mathcal W_{k,B}^{(J)}(A^{(k)})
  -\mathcal W_{k-1,B}^{(J)}(A^{(k)}).                    \tag{3.6}
\]
Then
\[
 \E[\mathfrak M_{k,B}\mid\cF_{k-1}]=0                 \tag{3.7}
\]
and pathwise
\[
 \mathfrak P_{k,B}
 =\mathcal W_{k,B}^{(J)}(A^{(k)})
  -\mathcal W_{k-1,B}^{(J)}(A^{(k-1)})
  -\mathfrak M_{k,B}.                                    \tag{3.8}
\]
The first two terms telescope, so
\[
 \boxed{
 \E\sum_{k=j_0}^J\mathfrak P_{k,B}
 =\E\left[
 \mathcal W_{J,B}^{(J)}(A^*)
 -\mathcal W_{j_0-1,B}^{(J)}(A^0)\right].}               \tag{3.9}
\]
This is an expectation identity.  The correction (3.6) remains pathwise.

Insert (6.7)--(6.8) into (6.4). \qed
