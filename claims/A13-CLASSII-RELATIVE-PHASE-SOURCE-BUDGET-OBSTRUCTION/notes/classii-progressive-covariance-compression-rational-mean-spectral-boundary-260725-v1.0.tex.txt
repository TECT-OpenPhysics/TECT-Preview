\section{Global progressive covariance compression}

Fix a finite cutoff $J$.  Let ${\cal U}=L^2([0,1];H_0)$ be the control
space, let $H_J$ be the finite-dimensional terminal Gaussian space, and let
\[
 T_Jv=\int_0^1K_J(t)v_t\,dt,\qquad
 C_J=T_JT_J^*=\int_0^1K_J(t)K_J(t)^*\,dt.
 \tag{2.1}
\]
No disjoint-range or one-shot assumption is imposed on $K_J(t)$.

\subsection*{Theorem 2.1 (pathwise global contraction)}

For every $v\in{\cal U}$,
\[
 \boxed{\|C_J^{\dagger/2}T_Jv\|_{H_J}^2
 \le\int_0^1\|v_t\|_{H_0}^2dt.}
 \tag{2.2}
\]
Here $\dagger$ is the Moore--Penrose inverse, so the left side is the
canonical Cameron--Martin norm on $\operatorname{Ran}C_J^{1/2}$.

\emph{Proof.}
The polar decomposition of $T_J$ can be written
\[
 T_J=C_J^{1/2}R_J,
 \tag{2.3}
\]
where $R_J:{\cal U}\to H_J$ is a contraction whose range lies in
$\overline{\operatorname{Ran}C_J}$.  Thus
$C_J^{\dagger/2}T_J=P_{\operatorname{Ran}C_J}R_J$ and has operator norm at
most one.  Squaring proves (2.2). \hfill$\square$

The theorem is deterministic and pathwise.  It therefore applies to a
random progressive $v$ after fixing the sample.  In particular, revisiting
the same terminal direction many times cannot create a multiplicity loss:
the same overlap appears in both $T_Jv$ and $C_J=T_JT_J^*$ and is normalized
by the pseudoinverse.

Under the A8 production symbol bound
\[
 \mathcal A(k)\succeq c_{\rm sym}\langle k\rangle^4I
 \tag{2.4}
\]
and the regulator multiplier bound $\|R_J\|\le M_R$, (2.2) yields
\[
 \boxed{\|T_Jv\|_{H^2}^2
 \le {M_R^2\over c_{\rm sym}}
 \int_0^1\|v_t\|^2dt.}
 \tag{2.5}
\]
This is the finite-cutoff Cameron--Martin embedding with constants uniform
in $J$.

\subsection*{Hilbert martingale and spatial ledgers}

Let ${\cal F}_j$ reveal the Gaussian roots in increasing order and put
\[
 d_jA=\mathbb E[A\mid{\cal F}_j]
      -\mathbb E[A\mid{\cal F}_{j-1}],\qquad A=T_Jv.
 \tag{2.6}
\]
Martingale differences are orthogonal in the Cameron--Martin Hilbert space,
hence
\[
 \sum_j\mathbb E\|d_jA\|_{\rm CM}^2
 \le\mathbb E\|A\|_{\rm CM}^2
 \le\mathbb E\int_0^1\|v_t\|^2dt.
 \tag{2.7}
\]
The same argument after the deterministic $H^2$ multiplier in (2.5) gives
\[
 \sum_j\mathbb E\|d_jA\|_{H^2}^2
 \le {M_R^2\over c_{\rm sym}}\mathbb E\int_0^1\|v_t\|^2dt.
 \tag{2.8}
\]
For an inhomogeneous Littlewood--Paley decomposition, every $0<s\le2$, and
roots $j\ge j_0$,
\[
 \boxed{
 \sum_{j,m}2^{2s(m-j)}
 \mathbb E\|\Delta_md_jA\|_2^2
 \le C_{s,j_0}{M_R^2\over c_{\rm sym}}
 \mathbb E\int_0^1\|v_t\|^2dt.}
 \tag{2.9}
\]
Indeed, $2^{-2sj}\|d_jA\|_{H^s}^2$ is bounded by a constant times
$\|d_jA\|_{H^2}^2$, and (2.8) sums the latter exactly once.  Restricting
(2.9) to $m\ge j+C$ gives the far gain
\[
 \sum_j\sum_{m\ge j+C}\mathbb E\|\Delta_md_jA\|_2^2
 \le C_{s,j_0}2^{-2sC}{M_R^2\over c_{\rm sym}}
 \mathbb E\int_0^1\|v_t\|^2dt.
 \tag{2.10}
\]

R-088 proved, for $0<s<1$,

Equation (2.5) removes its former one-shot restriction.  If

\section{Direct integrated Cartan Fourier trace}

Fix a generator $A$ and root $j$.  Sum the R-088 sequential physical shells
before taking any square.  Write
\[
 Z_j^-=w_j+A^{(\ell)},\qquad
 a^{[j]}=A^{(j)}-A^{(\ell)},\qquad
 Z_j^+=Z_j^-+a^{[j]}.
 \tag{3.1}
\]
For the production map $\Phi=\Phi_{A,j}$ define
\[
 c_{A,j}=\Phi(Z_j^+)-\Phi(Z_j^-)
 \tag{3.2}
\]
and, for each spatial coordinate $i$,
\[
 \begin{aligned}
 b_{A,j,i}={}&[D\Phi(Z_j^+)-D\Phi(Z_j^-)]^T\partial_iZ_j^-\\
 &+D\Phi(Z_j^+)^T\partial_i a^{[j]}.
 \end{aligned}
 \tag{3.3}
\]
The exact telescope and the product rule give
\[
 \boxed{\sum_{k\le j}T_{A,j,k}[v]=L_{A,j}v,\qquad
 (L_{A,j}v)_i=b_{A,j,i}^Tv+c_{A,j}^T\partial_iv.}
 \tag{3.4}
\]
This identity contains every cross term between different physical control
shells.  Ornstein--Uhlenbeck averaging preserves the first-order form; below
$b_{A,j,t}$ and $c_{A,j,t}$ denote the averaged coefficients.

Let the root covariance have Fourier eigenvectors
\[
 v_{p,\mu}(x)=\sqrt{\lambda_{j,p}}e^{ipx}e_\mu,
 \qquad p\in S_j\subset Q_j,
 \tag{3.5}
\]
and suppose
\[
 \Lambda_{0,j}:=\sum_{p,\mu}\lambda_{j,p}\le\kappa_02^{-j},
 \qquad
 \Lambda_{1,j}:=\sum_{p,\mu}\lambda_{j,p}|p|^2\le\kappa_12^j.
 \tag{3.6}
\]

\subsection*{Theorem 3.1 (exact trace and two-tail bound)}

For an output multiplier $\Pi_m$ and $m\ge j+3$,
\[
 \begin{aligned}
 \|\Pi_mL_{A,j,t}\Gamma_j^{1/2}\|_{\rm HS}^2
 ={}&\sum_{\mu,p}\lambda_{j,p}\sum_q|\pi_m(q)|^2\\
 &\quad\times\sum_i|\widehat b_{A,j,t,i}(q-p)e_\mu
       +ip_i\widehat c_{A,j,t}(q-p)e_\mu|^2.
 \end{aligned}
 \tag{3.7}
\]
If $\widetilde\Pi_m$ selects
$2^{m-2}<|r|_\infty\le2^{m+2}$, then
\[
 \boxed{
 \|\Pi_mL_{A,j,t}\Gamma_j^{1/2}\|_{\rm HS}^2
 \le2\Lambda_{0,j}\|\widetilde\Pi_mb_{A,j,t}\|_2^2
   +2\Lambda_{1,j}\|\widetilde\Pi_mc_{A,j,t}\|_2^2.}
 \tag{3.8}
\]

\emph{Proof.}
Apply (3.4) to each vector (3.5).  The Fourier coefficient at output $q$
is exactly the vector displayed in (3.7), so Parseval and summation over the
root basis give equality.  When $q$ is in shell $m$, $p$ is in shell $j$,
and $m\ge j+3$, the difference $q-p$ lies in the enlarged annulus.  Apply
$|x+y|^2\le2|x|^2+2|y|^2$, sum first in $q$, and then use (3.6).  This proves
(3.8). \hfill$\square$

The enlarged annuli have overlap at most five.  Summing (3.8) over
$m\ge j+C$ and restoring the R-084 prefactor $3/(40P)$ yields
\[
 \boxed{
 S_C^{\rm ctrl}\le {3\over4P}\sum_{A,j}\int_0^\infty e^{-2t}\mathbb E
 \left[\kappa_02^{-j}\|Q^{\rm coef}_{j,C-2}b_{A,j,t}\|_2^2
 +\kappa_12^j\|Q^{\rm coef}_{j,C-2}c_{A,j,t}\|_2^2\right]dt.}
 \tag{3.9}
\]
No output orthogonality and no per-$k$ quantity $q_k$ was used.

For $s>0$, Sobolev tail extraction reduces (3.9) to
\[
 S_C^{\rm ctrl}\le {3\over4P}2^{-2s(C-2)}\mathfrak E_s,
 \tag{3.10}
\]
where
\[
 \boxed{
 \mathfrak E_s=
 \sum_{A,j}\int_0^\infty e^{-2t}\mathbb E\left[
 \kappa_02^{-(2s+1)j}\|b_{A,j,t}\|_{\dot H^s}^2
 +\kappa_12^{(1-2s)j}\|c_{A,j,t}\|_{\dot H^s}^2
 \right]dt.}
 \tag{3.11}
\]
The remaining Cartan theorem is exactly
\[
 \mathfrak E_s\lesssim
 1+\mathbb E[X^{1/2}Y^{1/2}]
 \tag{3.12}
\]
for some $s>0$.  Equations (3.7)--(3.11) prove the direct integrated
reduction, not (3.12).
