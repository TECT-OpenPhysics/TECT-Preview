% evidence-anchor: theorem-2.1-affine-residual-douglas
\textbf{Theorem 2.1 (sharp affine-residual Douglas bound).}
Suppose there are constants
\[
 0\leq\rho<\sigma<1,\qquad M<\infty,\qquad C_0<\infty                \tag{2.3}
\]
uniform in cutoff, chart, refinement, and admissible $h$, with
\[
 \|C_{J,\pi}\|\leq\rho,
 \qquad \|E_{J,\pi}(h)\|\leq M,
 \qquad a_{J,\pi}\geq-C_0.                          \tag{2.4}
\]
Then
\[
\boxed{
 \cA_{J,\pi}(h)\geq
 -C_0-{\sigma^2M^2\over2(\sigma^2-\rho^2)}
 +{1-\sigma^2\over2}\|\cY_{J,\pi}(h)\|^2.}        \tag{2.5}
\]
The residual coefficient in (2.5) is sharp under norm data alone.

\textit{Proof.}
Put $y=\|\cY\|$ and $m=\|E\|$.  The triangle inequality and
$\|C\|\leq\rho$ give $\|\cU\|\leq\rho y+m$.  For
$\delta=\sigma^2-\rho^2>0$,
\[
\begin{split}
 &\sigma^2y^2+{\sigma^2\over\delta}m^2-(\rho y+m)^2\\
 &\hspace{20mm}={1\over\delta}(\delta y-\rho m)^2\geq0.
\end{split}                                                          \tag{2.6}
\]
Substitution in (2.1) proves (2.5).  Equality in (2.6) occurs at
$y=\rho m/\delta$, proving sharpness. \hfill$\square$

Write
\[
 X(h)=\E\sum_k\|h_k\|_2^2,
 \qquad Y_6(h)=\E\|Z_{K,h}\|_6^6.                  \tag{2.7}
\]
Since
\[
 \|\cY(h)\|^2=\|\Phi(h)\|^2+{9\over10}X(h)
                                  +{3\over10}Y_6(h),                  \tag{2.8}
\]
subtracting the stabilizers from (2.5) gives the exact consequence
\[
\boxed{
 \begin{split}
 \E V_J^{\rm ren}(Z_h)\geq{}&
 -{9\over20}\sigma^2X(h)-{3\over20}\sigma^2Y_6(h)\\
 &+{1-\sigma^2\over2}\|\Phi(h)\|^2
 -C_0-{\sigma^2M^2\over2(\sigma^2-\rho^2)}.
 \end{split}}                                                        \tag{2.9}
\]
Choose the comparison exponent $p=11/10$ and retain the Nelson
exponent $q=10/9$.  Then $q-p=1/90$ and
\[
 \epsilon_v={9\over20}\sigma^2<{9\over20}={1\over2q}
 <{5\over11}={1\over2p},
 \qquad {5\over11}-{9\over20}={1\over220},
 \qquad
 \epsilon_6={3\over20}\sigma^2<{3\over20}<{\gamma\over6}.         \tag{2.10}
\]
where the A1 value is $\gamma=81/50$ and $\gamma/6=27/100$.

% evidence-anchor: corollary-2.2-t050-sufficient-condition
