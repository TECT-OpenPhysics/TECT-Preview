\section{Complete random-$W$ score transfer}

Condition on $\cF_{\rm past}$.  Let $\xi\sim N(0,\Id_d)$, let $H$ be a real
Hilbert space, let $h:\R^d\to H$, and let
$W:\R^d\to\operatorname{Sym}_d$ be positive semidefinite.  Put
\[
 (\operatorname{div}W)_i=\sum_j\partial_jW_{ij},\qquad
 \operatorname{div}^2W=\sum_{i,j}\partial_{ij}W_{ij}.       \tag{1.1}
\]
The current-root Gaussian divergence and its second iterate are
\[
 (\delta_\xi W)_i=(W\xi-\operatorname{div}W)_i,              \tag{1.2}
\]
\[
 \boxed{
 \delta_\xi^2W
 =\xi^TW\xi-\Tr W-2\xi\cdot\operatorname{div}W
   +\operatorname{div}^2W.}                                 \tag{1.3}
\]

Define the complete signed second jet
\[
 \mathfrak F_W(h)=\E_\xi\left[
 \|D_\xi h\,W^{1/2}\|_{\HS}^2
 +\langle h,\Tr_{\R^d}(W D_\xi^2h)\rangle
 \mid\cF_{\rm past}\right].                                \tag{1.4}
\]

% evidence-anchor: theorem-1.1-random-w-double-divergence-score-transfer
\subsection*{Theorem 1.1 (random-covariance double-divergence transfer)}

Under the preceding domain and integrability assumptions,
\[
 \boxed{
 \mathfrak F_W(h)
 ={1\over2}\E_\xi[\|h\|^2\delta_\xi^2W
                   \mid\cF_{\rm past}].}                    \tag{1.5}
\]
Equivalently,
\[
 \boxed{
 \mathfrak F_W(h)=\E_\xi[
 \langle h,D_\xi h(W\xi-\operatorname{div}W)\rangle
 \mid\cF_{\rm past}].}                                     \tag{1.6}
\]

\emph{Proof.}
For $g=\|h\|^2$,
\[
 {1\over2}\partial_{ij}g
 =\langle\partial_i h,\partial_jh\rangle
  +\langle h,\partial_{ij}h\rangle.                          \tag{1.7}
\]
Contract (1.7) with $W_{ij}$ and use twice the adjoint relation between
$D_\xi$ and $\delta_\xi$.  This proves (1.5).  Applying the adjoint relation
once proves (1.6). \hfill$\square$

The naive frozen-coefficient comparator is
\[
 {1\over2}\E_\xi[
 \|h\|^2(\xi^TW\xi-\Tr W)\mid\cF_{\rm past}].              \tag{1.8}
\]
It is not obtained by actually freezing a random $W$.  The exact correction
between (1.5) and (1.8) is
\[
 \boxed{
 \mathcal R_W(h)=\E_\xi\left[
 \|h\|^2\left\{-\xi\cdot\operatorname{div}W
 +{1\over2}\operatorname{div}^2W\right\}
 \mid\cF_{\rm past}\right].}                               \tag{1.9}
\]
Thus the $DW,D^2W$ forest is not deleted; it is assembled into one raw
double-divergence coordinate.

% evidence-anchor: theorem-1.2-random-w-young-optimized-form-bound
\subsection*{Theorem 1.2 (Young-optimized conditional form cost)}

Let
\[
 K_W=\E_\xi[(\delta_\xi^2W)^2\mid\cF_{\rm past}],\quad
 X=\E_\xi[\|h\|^2\mid\cF_{\rm past}],\quad
 Y=\E_\xi[\|h\|^6\mid\cF_{\rm past}].                     \tag{1.10}
\]
Then
\[
 \boxed{
 |\mathfrak F_W(h)|\le {1\over2}K_W^{1/2}X^{1/4}Y^{1/4}.}   \tag{1.11}
\]
For every $q,\eta,\zeta>0$,
\[
 \boxed{
 -{q\over2}\mathfrak F_W(h)
 \ge-\eta X-\zeta Y
 -{q^2K_W\over128\sqrt{\eta\zeta}}.}                       \tag{1.12}
\]

\emph{Proof.}
Equation (1.5), Cauchy--Schwarz, and
$\E\|h\|^4\le X^{1/2}Y^{1/2}$ give (1.11).  Apply the optimized scalar inequality
\[
 A x^{1/4}y^{1/4}
 \le\eta x+\zeta y+{A^2\over8\sqrt{\eta\zeta}}             \tag{1.13}
\]
with $A=qK_W^{1/2}/4$. \hfill$\square$

If $W$ is strict-past measurable, then
\[
 K_W=2\|W\|_{\HS}^2.                                       \tag{1.14}
\]
At $q=10/9$, the random-score coefficient in (1.12) is $25/2592$;
(1.14) recovers the R-109 coefficient $25/1296$ exactly.

\subsection*{Corollary 1.3 (dimension-free chaos--Sobolev sufficient cost)}

Write the conditional chaos expansion
\[
 W=\sum_{n\ge0}I_n(f_n),\qquad
 f_n\in(\R^d)^{\odot n}\otimes(\R^d)^{\odot2}.             \tag{1.15}
\]
Full symmetrisation of the $n+2$ indices does not increase norm.  Since
\[
 (n+1)(n+2)=n(n-1)+4n+2,                                    \tag{1.16}
\]
one obtains
\[
 \boxed{
 K_W\le\E_\xi[
 \|D^2W\|^2+4\|DW\|^2+2\|W\|^2
 \mid\cF_{\rm past}].}                                    \tag{1.17}
\]
The norms in (1.17) are full tensor Hilbert norms.  Equality holds in one
dimension and whenever every chaos kernel is symmetric across all slots.

This is a sufficient production gate, not an accepted production estimate.
An arbitrary Boue--Dupuis control has only the declared source energy; no
uniform $DW,D^2W$ budget may be assumed.

\section{Static covariance norms do not control current-root rotation}

\subsection*{Proposition 2.1 (uniformly positive scalar no-go)}

Let $G\sim N(0,1)$, $0<\epsilon<1$, $\tau\in\{+1,-1\}$, and
\[
 h_M(G)=M^{-1/2}\sin(MG),\qquad
 w_M^\tau(G)=1+\tau\epsilon\cos(2MG).                       \tag{2.1}
\]
Then $1-\epsilon\le w_M^\tau\le1+\epsilon$, but
\[
 \boxed{
 \mathfrak F_{w_M^\tau}(h_M)
 =M\left[e^{-2M^2}+{\tau\epsilon\over2}(1+e^{-8M^2})\right]
 \sim {\tau\epsilon\over2}M.}                              \tag{2.2}

Meanwhile
\[
 X_M={1-e^{-2M^2}\over2M}\longrightarrow0,                 \tag{2.3}

\[
 Y_M={10-15e^{-2M^2}+6e^{-8M^2}-e^{-18M^2}\over32M^3}

The correct score cost is
\[
 \boxed{
 \begin{aligned}
 K_{w_M^\tau}={}&2+4\tau\epsilon e^{-2M^2}\cr
 &+\epsilon^2\{1+8M^2+8M^4
 +(1-8M^2+8M^4)e^{-8M^2}\},
 \end{aligned}}                                             \tag{2.5}

and hence $K_{w_M^\tau}\sim8\epsilon^2M^4$.

\section{Trace-corrected diagonal-to-decoupled interpolation}

Condition again on the strict past.  Let $X,X'$ be independent standard
Gaussians in $\R^d$, let
\[
 Z_\rho=\rho X+\sqrt{1-\rho^2}\,X',\qquad0\le\rho\le1,       \tag{3.1}
\]
and set
\[
 y(x,z)=b+Ax+R(z),\qquad S=AA^*.                             \tag{3.2}
\]
Write $a_i=Ae_i$, $r_i=\partial_iR$, and
\[
 c(z)=\Tr(A^*DR(z))=\sum_i\langle a_i,r_i(z)\rangle.        \tag{3.3}
\]

The family of traces is compatible precisely when an actual decoupled
trace/forest $\delta_0$ satisfies
\[
 \boxed{\delta_\tau=\delta_0+2c},                            \tag{3.4}
\]
and one transports
\[
 \delta_\rho=\delta_0+2\rho c.                              \tag{3.5}
\]
The physical use of (3.4) is a theorem obligation: $\delta_0$ may not be
defined formally after deleting R-063 owners.

Define
\[
 \ell_\rho(x,z)={1\over2}\|y(x,z)\|^2-{1\over2}\Tr S
                 -{1\over2}\delta_\rho(z),                  \tag{3.6}
\]
\[
 \Phi(\rho)=\log\E_{\gamma_\rho}e^{-q\ell_\rho},           \tag{3.7}
\]
where $\gamma_\rho$ is the joint Gaussian law of $(X,Z_\rho)$.
For Theorem 3.1, take $q\ge0$, assume $R$ and $\delta_0$ are cylindrical
$C^3$ functions, and assume the exponential moments of $\ell_\rho$ together
with every derivative appearing in (3.8)--(3.15) are finite uniformly on
$0\le\rho\le1$.  The identities are first proved under these hypotheses;
standard approximation may be used only when it preserves these bounds.

% evidence-anchor: theorem-3.1-trace-corrected-diagonal-interpolation
\subsection*{Theorem 3.1 (exact diagonal interpolation)}

Let
\[
 p_\rho=A^*y,\qquad
 s_\rho=DR^*y-{1\over2}\nabla\delta_\rho,\qquad
 \Gamma_\rho=p_\rho\cdot s_\rho.                            \tag{3.8}
\]
Then
\[
 \boxed{{d\over d\rho}\E_{\gamma_\rho}\ell_\rho=0},       \tag{3.9}
\]
and, for the tilted probability law
\[
 d\nu_\rho={e^{-q\ell_\rho}\over
                   \E_{\gamma_\rho}e^{-q\ell_\rho}}
             d\gamma_\rho,                                 \tag{3.10}
\]
one has
\[
 \boxed{\Phi'(\rho)=q^2\E_{\nu_\rho}\Gamma_\rho.}          \tag{3.11}
\]
In components,
\[
 \Gamma_\rho=\sum_j\langle y,a_j\rangle
 \left\{\langle y,r_j\rangle
 -{1\over2}\partial_j\delta_0
 -\rho\sum_i\langle a_i,r_{ij}\rangle\right\}.             \tag{3.12}
\]

At the decoupled endpoint, with $K_q=(\Id+qS)^{-1}$,
\[
 \boxed{
 \begin{aligned}
 \Phi(0)={}&-{1\over2}\log\det{}_2(\Id+qS)\cr
 &+\log\E_z\exp\left\{{q\over2}\delta_0(z)
 -{q\over2}\langle b+R(z),K_q(b+R(z))\rangle\right\}.
 \end{aligned}}                                             \tag{3.13}
\]
Consequently,
\[
 \boxed{\Phi(1)=\Phi(0)+q^2\int_0^1
                  \E_{\nu_\rho}\Gamma_\rho\,d\rho.}       \tag{3.14}
\]

\emph{Proof.}
Price's formula gives
\[
 {d\over d\rho}\E_{\gamma_\rho}f
 =\E_{\gamma_\rho}\sum_i\partial_{x_i}\partial_{z_i}f.
                                                                    \tag{3.15}
\]
Here $\partial_\rho\ell_\rho=-c$ and
$\sum_i\partial_{x_i}\partial_{z_i}\ell_\rho=c$, proving (3.9).
Applying the same operator to $e^{-q\ell_\rho}$ cancels both linear trace
terms and leaves $q^2\Gamma_\rho e^{-q\ell_\rho}$, proving (3.11).
At $\rho=0$, integrate the independent $X$ square to obtain (3.13), then
integrate (3.11). \hfill$\square$

The theorem converts the diagonal problem to two explicit gates:
\[
 \log\E\exp\left\{{q\over2}\delta_0
 -{q\over2}\langle b+R,K_q(b+R)\rangle\right\}\le r_0,     \tag{3.16}
\]
\[
 \int_0^1\E_{\nu_\rho}\Gamma_\rho\,d\rho\le G.           \tag{3.17}
\]
If both costs are predictable, then
\[
 \Phi(1)\le{q^2\over4}\|S\|_{\HS}^2+r_0+q^2G.             \tag{3.18}
\]
No accepted authority proves (3.16)--(3.17) for the production cluster.
In particular, (3.12) differentiates the decoupled trace and mixed trace and
therefore exposes the second-/third-jet forest rather than removing it.

\section{The smallest stationary scalar cross-output model}

Work on the normalized circle with $k=1$ and
\[
 X(x)=A+a\cos x+b\sin x+c\cos2x+d\sin2x,                    \tag{5.1}
\]
where $(a,b)\sim N(0,v\Id_2)$, $(c,d)\sim N(0,w\Id_2)$, and
the pairs are independent.  Put
\[
 z={a-ib\over2},\quad u={c-id\over2},\quad
 R=|z|^2,\quad S=|u|^2,\quad C=\operatorname{Re}(z^2\bar u),
 \quad\gamma=v+4w.                                          \tag{5.2}
\]
Assume first that $v,w>0$.  Then $2R/v$ and $2S/w$ are independent unit
exponentials and
$|C|\le R\sqrt S$.
All polynomial moment identities below extend continuously to $v,w\ge0$;
the zero-variance endpoints are interpreted by that extension, not by
dividing by $v$ or $w$.  This is a stationary scalar two-frequency model for
the first cross-output obstruction.  No full A1 embedding is claimed here.

For $J=X\partial_xX$, the positive output coefficients are
\[
 {\widehat J(1)\over i}=Az+\bar zu,\quad
 {\widehat J(2)\over i}=z^2+2Au,\quad
 {\widehat J(3)\over3i}=zu,\quad
 {\widehat J(4)\over2i}=u^2.                                \tag{5.3}
\]
All conjugate outputs and their cross covariances are retained.

Let $G$ be an independent copy of the random Fourier part and set
$Y_X=X\partial_xG$.  If
\[
 x_0=A,\quad x_{\pm1}=z,\bar z,\quad
 x_{\pm2}=u,\bar u,\qquad
 q_{\pm1}=v/2,\quad q_{\pm2}=w/2,                            \tag{5.4}
\]
the realized whole-output covariance is
\[
 \boxed{
 S_X(n,m)=\sum_{r=\pm1,\pm2}r^2q_r
 x_{n-r}\overline{x_{m-r}}.}                                \tag{5.5}
\]
In particular,
\[
 \Tr S_X=\gamma(A^2+2R+2S).                                 \tag{5.6}
\]

% evidence-anchor: theorem-5.1-complete-k2k-packet-moments
\subsection*{Theorem 5.1 (exact complete $k{:}2k$ packet and square margin)}

The complete covariance-normal packet
\[
 P={1\over2}\{\|J\|_2^2-\Tr S_X\}                          \tag{5.7}
\]
is exactly
\[
 \boxed{
 \begin{aligned}
 P={}&A^2R+4A^2S-\frac{\gamma A^2}{2}+6AC\cr
 &+R^2+10RS+4S^2-\gamma(R+S).
 \end{aligned}}                                             \tag{5.8}
\]
It is centered.  Moreover, with
\[
 H=\E\|S_X\|_{\HS}^2,                                      \tag{5.9}
\]
one has
\[
 \boxed{
 \begin{aligned}
 H={}&A^4\left({v^2\over2}+8w^2\right)\cr
 &+A^2(v^3+10v^2w+16vw^2+16w^3)\cr
 &+{5\over4}v^4+v^3w+25v^2w^2+16vw^3+20w^4,
 \end{aligned}}                                             \tag{5.10}
\]
\[
 \boxed{
 \begin{aligned}
 \Var(P)={1\over4}\{&A^4v^2+16A^4w^2
 +2A^2v^3+20A^2v^2w\cr
 &+32A^2vw^2+32A^2w^3
 +2v^4+2v^3w\cr
 &+42v^2w^2+32vw^3+32w^4\}.
 \end{aligned}}                                             \tag{5.11}
\]
The square-first leading margin is the positive identity
\[
 \boxed{
 H-2\Var(P)={v^4+16v^2w^2+16w^4\over4}.}                   \tag{5.12}
\]
It is strict when $v+w>0$ and zero only for the degenerate cluster.

\emph{Proof.}
Fourier convolution gives (5.3), and Parseval gives (5.8).  Write the four
columns of the map $G\mapsto X\partial_xG$ as $\ell_r$; then
\[
 S_X=\sum_r\ell_r\otimes\ell_r,\qquad
 \|S_X\|_{\HS}^2=\sum_{r,s}|\langle\ell_r,\ell_s\rangle|^2. \tag{5.13}
\]
Wick expansion gives (5.10)--(5.11), and subtraction gives (5.12).
The primary certificate derives the Gram matrix by exact trigonometric
integration; the independent certificate reconstructs it from Fourier
convolution without importing the primary. \hfill$\square$

For $q\downarrow0$ from the right,
\[
 \log\E e^{-qP}={q^2\over2}\Var(P)-{q^3\over6}\E P^3+O(q^4).
                                                                    \tag{5.14}
\]
All coefficients in the exact polynomial $\E P^3$ are positive for
$A^2,v,w\ge0$.  Thus (5.12) and the cubic term both favor the live target
\[
 \boxed{
 \log\E e^{-qP}\stackrel{?}{\le}{q^2\over4}H,\qquad q\ge0.} \tag{5.15}
\]
This is only a one-sided small-$q$ conclusion.  A positive exponential
moment of $P$ need not exist, so no two-sided analytic neighborhood is being
claimed.  Four declared finite points have positive margins at both
$24\times24$ and $48\times48$ Gauss--Laguerre/Bessel orders in the primary
JSON.  This is only a two-order sign-stability check: the numerical margins
are not claimed exact or converged, and the finite scan does not prove
