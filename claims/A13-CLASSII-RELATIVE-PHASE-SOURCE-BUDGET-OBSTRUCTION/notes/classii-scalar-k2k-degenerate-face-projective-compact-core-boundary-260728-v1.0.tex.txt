\section{All-$q$ theorem on both degenerate frequency faces}

% evidence-anchor: theorem-2.1-degenerate-frequency-faces-all-q
\subsection*{Theorem 2.1 (exact all-$q$ closure for $vw=0$)}

For every $A\in\R$, $v,w\ge0$ with $vw=0$, and every $q\ge0$,
\[
 \boxed{\log\E e^{-qP}\le {q^2\over4}H.}                 \tag{2.1}
\]

\emph{Proof on $w=0$.}
Assume $v>0$, let
\[
 \alpha={A^2\over v},\qquad X={2R\over v}\sim\operatorname{Exp}(1),
 \qquad s={qv^2\over4}.                                   \tag{2.2}
\]
Then
\[
 P={v^2\over4}Z_\alpha,\qquad
 Z_\alpha=2\alpha(X-1)+W,\qquad W=X^2-2X\ge-1,           \tag{2.3}
\]
and (2.1) is exactly
\[
 \log\E e^{-sZ_\alpha}
 \le(2\alpha^2+4\alpha+5)s^2.                            \tag{2.4}
\]

Tilt first by the centered linear exponential.  With
\[
 \ell=1+2\alpha s,                                        \tag{2.5}
\]
one has the exact identity
\[
 \log\E e^{-sZ_\alpha}
 =2\alpha s-\log\ell
  +\log\E_{\operatorname{Exp}(\ell)}e^{-sW}.              \tag{2.6}
\]
The first two terms obey
\[
 2\alpha s-\log(1+2\alpha s)\le2\alpha^2s^2.             \tag{2.7}
\]
Under $\operatorname{Exp}(\ell)$, direct moments give
\[
 \E_\ell W=-{2(\ell-1)\over\ell^2},\qquad
 \Var_\ell(W)={4(\ell^2-4\ell+5)\over\ell^4}\le8.     \tag{2.8}
\]
The last inequality follows from
\[
 8\ell^4-4(\ell^2-4\ell+5)
 =4(\ell-1)(2\ell^3+2\ell^2+\ell+5)\ge0.                \tag{2.9}
\]

If $s\ge(4\alpha+5)^{-1}$, the floor $W\ge-1$ gives
\[
 \log\E_\ell e^{-sW}\le s\le(4\alpha+5)s^2.           \tag{2.10}
\]
Together with (2.7), this proves (2.4) on the large-$s$ branch.

If $s\le(4\alpha+5)^{-1}\le1/5$, put
$Y=\E_\ell W-W$.  Then $\E_\ell Y=0$, $Y\le1$, and
$\E_\ell Y^2\le8$.  The one-sided exponential estimate gives
\[
\begin{aligned}
 \log\E_\ell e^{-sW}
 &=-s\E_\ell W+\log\E_\ell e^{sY}\\
 &\le {4\alpha s^2\over\ell^2}+8(e^s-1-s)\\
 &\le\left(4\alpha+{30\over7}\right)s^2
 <(4\alpha+5)s^2.
\end{aligned}                                              \tag{2.11}
\]
Here the last estimate uses
$e^s-1-s\le s^2/[2(1-s/3)]$ for $0\le s\le1/5$.
Equations (2.7) and (2.11) prove (2.4) on the small-$s$ branch.

\emph{The $v=0$ face.}
If $w>0$, set $X=2S/w$ and $\alpha=A^2/w$.  Then
\[
 P=w^2 Z_\alpha,\qquad
 H=w^4(8\alpha^2+16\alpha+20).                            \tag{2.12}
\]
With $s=qw^2$, inequality (2.1) is again exactly (2.4).  Finally,
$v=w=0$ is trivial. \hfill$\square$

\paragraph{Boundary closed form.}
On $w=0$, direct integration also gives, with $t=qv^2$ and
$\beta=1+t(\alpha-1)/2$,
\[
\boxed{
 \log M_0(\alpha,t)
 ={t\alpha\over2}+{1\over2}\log{\pi\over t}
 +{\beta^2\over t}
 +\log\erfc\!\left({\beta\over\sqrt t}\right).}        \tag{2.13}
\]
The exact rational fixture $\alpha=27$, $t=1/9000$ has
\[
 {t^2h\over4}-\log M_0
 =2.1686752950311231951453702567\ldots\times10^{-9}>0,     \tag{2.14}
\]
and ratio $0.9982109464147929\ldots$.  This demonstrates the near-tight
large-amplitude/small-$q$ corner; it is not needed in the proof of Theorem 2.1.

\section{Large-amplitude projective boundary}

Write

\]
where

 L={T-1\over2}+2r(U-1),\qquad

% evidence-anchor: theorem-3.1-projective-large-amplitude-boundary
\subsection*{Theorem 3.1 (positive projective limit)}

Fix $x,r\ge0$ and set $t=x/a$.  Then
\[
 M(a,r,x/a)\longrightarrow
 M_\infty(x,r)
 ={e^{x(1+4r)/2}\over(1+x/2)(1+2rx)}.                    \tag{3.4}
\]
The limiting target gap is
\[
\boxed{
 D_0(x,r)=d(x/2)+d(2rx),\qquad
 d(y)={y^2\over2}-y+\log(1+y)\ge0.}                      \tag{3.5}
\]
Indeed,
\[
 d'(y)={y^2\over1+y}\ge0,\qquad d(0)=0.                 \tag{3.6}
\]
Thus $D_0(x,r)>0$ for $x>0$. \hfill$\square$

% evidence-anchor: proposition-3.2-positive-first-projective-correction
\subsection*{Proposition 3.2 (positive first inverse-amplitude correction)}

For fixed $x,r$,
\[
 {x^2h\over4a^2}-\log M(a,r,x/a)
 =D_0(x,r)+{D_1(x,r)\over a}+O(a^{-2}),                   \tag{3.7}
\]
where
\[
 \boxed{D_1(x,r)={x^3\Pi(x,r)\over
 4(x+2)^2(1+2rx)^2}>0}                                    \tag{3.8}
\]
for $x>0$, and
\[
\begin{aligned}
\Pi={}&64r^5x^3+256r^5x^2+256r^5x
 +64r^4x^3+320r^4x^2+512r^4x+256r^4\\
&+40r^3x^3+224r^3x^2+352r^3x+128r^3
 +4r^2x^3+56r^2x^2+172r^2x+112r^2\\
&+4rx^2+26rx+38r+x+4.
\end{aligned}                                              \tag{3.9}
\]

\emph{Proof.}
Under the limiting tilted product law, $T$ and $U$ remain independent
exponentials with rates $1+x/2$ and $1+2rx$.  Since the phase mean of $B$
vanishes,
\[
 \log M=\log M_\infty+{1\over a}
 \E_x\left[{x^2\over2}B^2-xQ\right]+O(a^{-2}).            \tag{3.10}
\]
Here the remainder is analytic rather than formal.  In the unscaled $v=1$
radial variables, for $a\ge20\gamma$,
\[
 a+10S-6\sqrt{aS}\ge {a\over10},
 \qquad
 p_-\ge R^2+{a\over20}R+4S^2+{79a\over20}S
              -{\gamma a\over2}.                          \tag{3.10a}
\]
At $t=x/a$ this lower bound gives an $a$-uniform exponential dominator for
the integrand and for every polynomial factor used below.  With
$\varepsilon=a^{-1/2}$ the phase-averaged integrand is even in
$\varepsilon$ (shift $\phi$ by $\pi$), so dominated Taylor expansion through
$\varepsilon^2$ has remainder $O(\varepsilon^4)=O(a^{-2})$ for fixed $x,r$.
Insert the exponential moments $\E T^n=n!/(1+x/2)^n$ and
$\E U^n=n!/(1+2rx)^n$, subtract the linear coefficient of $h$, and clear the
positive denominators.  The result is (3.8)--(3.9), whose coefficients are
all positive. \hfill$\square$

The leading floor-to-$\sqrt h$ shape satisfies
\[
 2+32r^2-(1+4r)^2=(4r-1)^2\ge0.                           \tag{3.11}
\]
The unique saturation ratio is $r=1/4$, but then
\[
 M_\infty(x,1/4)={e^x\over(1+x/2)^2},\qquad
 D_0(x,1/4)=2d(x/2)>0.                                    \tag{3.12}
\]

Proposition 3.2 is a pointwise projective expansion.  A uniform remainder
bound is still required before one may replace the entire large-$a$ region by
a finite certified threshold.  This boundary is explicit in the next-action
list and prevents an overclaim of compactness already completed.

\section{Exact phase floor, large-$q$ cutoff, and rigorous tails}

Use the normalized $v=1$ variables $R=T/2$, $S=rU/2$ and put $y=\sqrt S$.
Minimizing the phase gives
\[
\begin{aligned}
 p_-(R,S)={}&R^2+10RS+4S^2+(a-\gamma)R+(4a-\gamma)S\\
 &-6\sqrt a\,R\sqrt S-{\gamma a\over2},\qquad
 \gamma=1+4r.
\end{aligned}                                              \tag{4.1}
\]

% evidence-anchor: theorem-4.1-phase-minimum-high-q-reduction
\subsection*{Theorem 4.1 (finite algebraic floor and exact high-$q$ cutoff)}

For fixed $y$, the minimizing radial coordinate is
\[
 R_*(y)={1\over2}
 [\gamma-a-10y^2+6\sqrt a\,y]_+.                          \tag{4.2}
\]
Consequently
\[
\boxed{
 p_* =\min_{y\ge0}\left[
 4y^4+(4a-\gamma)y^2-{\gamma a\over2}
 -{1\over4}[\gamma-a-10y^2+6\sqrt a\,y]_+^2
 \right].}                                                \tag{4.3}
\]
The candidate set consists of $y=0$, the roots of the positive-part bracket,
the inactive quartic vertex, and the nonnegative roots of
\[
\boxed{
 84y^3-90\sqrt a\,y^2+(20a-8\gamma)y
 +3\sqrt a(\gamma-a)=0.}                                  \tag{4.4}
\]
Since $p\ge p_*$,
\[
 \log M(a,r,t)\le-tp_*.
\]
Therefore
\[
 \boxed{t\ge{4(-p_*)\over h}\quad\Longrightarrow\quad
 \log M(a,r,t)\le {t^2h\over4}.}                         \tag{4.5}
\]
Thus no fixed shape can fail at arbitrarily large $t$. \hfill$\square$

% evidence-anchor: theorem-4.2-bessel-and-factorized-tail-majorants
\subsection*{Theorem 4.2 (two globally integrable majorants)}

First, the modified Bessel function obeys
\[
\boxed{\log I_0(z)\le\sqrt{4+z^2}-2,\qquad z\ge0.}       \tag{4.6}
\]
To prove it, put $g=I_1/I_0$ and $h=z/\sqrt{z^2+4}$.  The Riccati identity is
$g'=1-g/z-g^2$.  At the origin,
\[
 g={z\over2}-{z^3\over16}+{z^5\over96}+O(z^7),\qquad
 h={z\over2}-{z^3\over16}+{3z^5\over256}+O(z^7),
\]
so $h-g=z^5/768+O(z^7)>0$ initially.  At a first positive contact,
\[
 g'-h'={4\sqrt{z^2+4}-z^2-8\over(z^2+4)^{3/2}}<0,         \tag{4.7}
\]
because
\[
 (z^2+8)^2-16(z^2+4)=z^4>0.                               \tag{4.8}
\]
An upward first crossing is impossible, so $g\le h$.  Integrate from zero to
obtain (4.6).  This bound has the correct quadratic/quartic expansion at zero
and linear growth at infinity.

Second, let $b=3\sqrt{ar/2}\,T\sqrt U$ and let $p_-=p_0-b$.  Exact square
completion gives
\[
\boxed{
 p_-\ge {T^2\over16}+{r^2U^2\over2}-C(a,r),}              \tag{4.9}
\]
where
\[
 C(a,r)={a\gamma\over2}+(\gamma-a)_+^2
 +{1\over2}\left(7a+{\gamma\over2}\right)^2.            \tag{4.10}
\]
Indeed, after subtracting the right side of (4.9), the low-$a$ branch is the
sum
\[
 \left({T\over2\sqrt2}-3\sqrt{arU}\right)^2
 +\left({T\over4}-(\gamma-a)\right)^2
 +{1\over2}\left(rU-7a-{\gamma\over2}\right)^2
 +{5\over2}rTU,                                           \tag{4.11}
\]
and on $a\ge\gamma$ the middle square is replaced by
$T^2/16+(a-\gamma)T/2$.

Since $I_0(tb)\le e^{tb}$, the exact integrand in (1.10) is at most
\[
 e^{tC(a,r)}e^{-T-tT^2/16}e^{-U-tr^2U^2/2}.               \tag{4.12}
\]
Every rectangular tail is therefore bounded by products of explicit
\[
 \int_L^\infty e^{-z-cz^2}dz
 ={\sqrt\pi\over2\sqrt c}e^{1/(4c)}
 \erfc\left(\sqrt cL+{1\over2\sqrt c}\right).             \tag{4.13}
\]
Equations (4.3)--(4.13) provide the algebraic high-$t$ and radial-tail pieces
of a finite-box certificate.  Completion still requires a uniform projective
remainder together with uniform local-$t$ and two-chart face patching as
$r\to0$ and $r\to\infty$. \hfill$\square$

\section{Failed routes and their exact boundaries}

% evidence-anchor: proposition-5.1-separated-conditional-domination-nogos
\subsection*{Proposition 5.1 (separated conditional domination fails)}

Three direct separations cannot prove the mixed theorem.

\paragraph{Quadratic Bessel envelope.}
The valid inequality $I_0(z)\le e^{z^2/4}$ inserts
\[
 +9q^2A^2R^2S                                             \tag{5.1}
\]
into the exponent.  Along $R=S=L$, this is positive cubic growth, while the
original coercive part is only $-15qL^2$.  The proposed upper integral is
infinite whenever $qA\ne0$.  The sharp linear-growth bound (4.6) is required.

\paragraph{Dropping the $S$ quartic.}
The elementary transform
$\int_0^\infty e^{-\alpha S}I_0(c\sqrt S)dS$ requires
\[
 \alpha={2\over w}+q(4A^2-\gamma+10R)>0.                  \tag{5.2}
\]
At $v=w=q=1$, $A=R=0$, it equals $-3$.  Thus one cannot discard $4qS^2$
before applying that transform.

\paragraph{Conditional scalar tensorization.}
Conditioning on $S$ creates the effective scalar coefficient
\[
 \alpha_{\rm eff}={A^2+10S-4w\over v},                    \tag{5.3}
\]
which is unbounded below across mixed shapes.  For
$\alpha_{\rm eff}=-B$,
\[
 Z_{-B}=(X-B-1)^2-(B^2+1).                                \tag{5.4}
\]
For fixed $0<s<1/2$, exact completion of the one-dimensional integral gives
\[
 \log\E e^{-sZ_{-B}}=sB^2-B+O(1),                         \tag{5.5}
\]
whereas blindly extending the positive-$\alpha$ proxy would give
\[
 (2B^2-4B+5)s^2=2s^2B^2+O(B).                             \tag{5.6}
\]
Because $s-2s^2>0$, (5.6) is false for large $B$.  Theorem 2.1 may not be
applied after this conditioning. \hfill$\square$

 \texttt{NG-2026-07-28-A13-K2K-CONDITIONAL-SCALAR-TENSORIZATION}.
\end{gathered}                                             \tag{5.7}
\]

\end{aligned}                                              \tag{5.10}
\]
Independent tensor Gauss--Laguerre orders 96 and 128 retain the same negative
