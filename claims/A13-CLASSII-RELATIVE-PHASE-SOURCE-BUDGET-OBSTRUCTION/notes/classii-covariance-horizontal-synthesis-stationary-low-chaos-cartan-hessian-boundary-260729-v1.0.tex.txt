been reconstructed: whether the realized covariance quotient controls the
physical endpoint norms, whether the R-118 quadratic multiplier theorem

\section*{2. Round one: exact six-current production frame}

Let $S_A=S_A^*$, $A=1,2,3$, be the fixed six-real embedded Pauli generators,
and write
\[
 m_A(u)=u^TS_Au,\qquad \rho(u)=u^Tu,\qquad

\]
where $e>0$ is the rational floor.  The A1 coefficient block is built from
the two one-forms
\[
 J_A=\dd m_A,
 \qquad K_A=\dd m_A-q_A\dd\rho.
\]
Its scalar weights are
\[
 a=\frac9{500P},\qquad b=\frac3{400P},\qquad
 c=\frac3{320P},\qquad P=4+10^{-12}.
\]

\subsection*{Theorem 2.1 (exact positive diagonalization)}

Set
\[
 c_0=\frac3{250P},\qquad c_1=\frac{243}{8000P},
 \qquad \alpha=\frac59,
\]
and
\[
 L_A=(1-\alpha)J_A+\alpha K_A
     =\dd m_A-\alpha q_A\dd\rho.
\]
Then, identically,
\[
 aJ_A^2+2bJ_AK_A+cK_A^2=c_0J_A^2+c_1L_A^2.       \tag{2.1}
\]
Indeed,
\[
 c_0+c_1(1-\alpha)^2=a,\qquad
 c_1\alpha(1-\alpha)=b,\qquad c_1\alpha^2=c.       \tag{2.2}
\]
Consequently the complete covariance coefficient is
\[
 B(u)=\sum_{A=1}^3
 \left(c_0p_Ap_A^T+c_1\ell_A\ell_A^T\right),       \tag{2.3}
\]
where
\[
 p_A=2S_Au,\qquad \ell_A=2(S_A-\alpha q_A\Id)u.    \tag{2.4}
\]
This is an exact six-current factorization; it is not a scalar-row proxy.

The fixed generators obey $S_A^2=P_{\rm dbl}$ on the active doublet and are
selfadjoint.  Hence
\[
 \sum_{A=1}^3|S_A|=3P_{\rm dbl}\preceq3\Id,        \tag{2.5}

These constants are derived from the A1 manifest rather than inserted from a

\section*{3. Round two: covariance-horizontal endpoint synthesis}

Let $\cS=\bigoplus_k\cS_k$ be the source Cameron--Martin space of a fixed
R-104/R-093 chart and let
\[
 L_\pi h=\sum_k S_kh_k,\qquad
 C_J=L_\pi L_\pi^*=\sum_kS_kS_k^*.                 \tag{3.1}
\]
For $z\in\Ran L_\pi$, define the realized quotient norm
\[
 \|z\|_{q,L_\pi}^2=\|C_J^{\dagger/2}z\|^2.        \tag{3.2}
\]

\subsection*{Theorem 3.1 (lossless quotient and physical $H^2$ control)}

For every source realization $h$ and $z=L_\pi h$,
\[
 \|z\|_{q,L_\pi}^2
 =\|P_{(\Ker L_\pi)^\perp}h\|_{\cS}^2
 \le \|h\|_{\cS}^2.                               \tag{3.3}
\]
For the regulated A1 covariance, mode by mode,
\[
 C_J(k)=|r_J(k)|^2\mathcal A(k)^{-1},\qquad
 \mathcal A(k)\succeq c_{\rm sym}\langle k\rangle^4\Id,
 \qquad |r_J(k)|\le M_R.                            \tag{3.4}
\]
It follows on $\Ran C_J$ that
\[
 \boxed{
 \|L_\pi h\|_{H^2}^2
 \le \frac{M_R^2}{c_{\rm sym}}
       \|L_\pi h\|_{q,L_\pi}^2
 \le \frac{M_R^2}{c_{\rm sym}}\|h\|_{\cS}^2.}    \tag{3.5}
\]
For the certified contractive regulator family,
\[
 M_R=1,\qquad c_{\rm sym}=0.10836453058711468\ldots,
 \qquad C_{\rm CM}=c_{\rm sym}^{-1}
 =9.22811176850986\ldots.                           \tag{3.6}
\]

\paragraph{Proof.}
The Moore--Penrose identity in (3.3) is the standard polar-decomposition
identity for $L_\pi$.  It is used only to evaluate the quotient norm; the
control is not replaced by a noncausal minimal representative.  On a Fourier
mode in the covariance range, (3.4) gives
\[
 C_J(k)^\dagger\succeq
 \frac{c_{\rm sym}}{M_R^2}\langle k\rangle^4\Id.
\]
Summation over modes proves (3.5).  The statement is independent of the
number of visits and of overlap among their ranges. \hfill$\square$

\subsection*{Lemma 3.2 (sharp two-endpoint $L^6$ bridge)}

Writing $Z_h=X_J+z$,
\[
 \boxed{
 \|z\|_6^6\le32\left(\|Z_h\|_6^6+\|X_J\|_6^6\right).} \tag{3.7}
\]
This is the pointwise inequality $|a-b|^6\le2^5(|a|^6+|b|^6)$, integrated
over space.  Equality is attained for $b=-a$, so $32$ cannot be reduced in
this two-term inequality.

\subsection*{Corollary 3.3 (once-only R-118 payment for fixed matrices)}

Let $A_0=M_R^2/c_{\rm sym}$.  Substituting (3.5)--(3.7) into the R-118
operator-valued quadratic estimate gives, for fixed selfadjoint generators
with $\sum_r|K_r|\preceq K_0\Id$,
\[
\begin{split}
 |\langle Q,\Pi_Kz\rangle|
 \le{}&\varepsilon\|h\|_{\cS}^2
 +\varepsilon_6\bigl(\|Z_h\|_6^6+\|X_J\|_6^6\bigr)\\
 &+C\sqrt{32}\,K_0^{10/3}A_0^{11/6}
 \varepsilon^{-11/6}\varepsilon_6^{-1/2}
 \|Q\|_{H^{-11/10}}^{10/3}.
\end{split}                                         \tag{3.8}
\]
Thus horizontal synthesis itself produces neither an inverse singular-value
loss nor a visit-count loss.  Equation (3.8) is a component theorem, not yet
the complete production inequality: the owner-complete $Q$ and generator
decomposition must still be proved.

\subsection*{Method boundary 3.4}

\section*{4. Round three: spatially varying multiplier theorem}

For selfadjoint matrix fields $K_r(x)$ assume, for every multi-index
$|\gamma|\le2$,
\[
 \sup_x\sum_r
 |\langle v,\partial^\gamma K_r(x)w\rangle|
 \le K_\gamma\|v\|\|w\|,                           \tag{4.1}
\]
and put
\[
 K_{\le2}=K_0+\sum_{1\le|\gamma|\le2}K_\gamma.
\]
Let $X=\|a\|_{H^2}^2$ and $Y=\|a\|_6^6$.

\subsection*{Theorem 4.1 (two-derivative multiplier extension)}

The centered quadratic output $\Pi_K(a)$ satisfies
\[
 \|\Pi_K(a)\|_2\le CK_0Y^{1/3},
 \qquad
 \|\Pi_K(a)\|_{H^2}\le CK_{\le2}X.                \tag{4.2}
\]
For $\kappa=1/10$, interpolation at
$\theta=(1+\kappa)/2=11/20$ yields
\[
 \|\Pi_K(a)\|_{H^{11/10}}
 \le C K_0^{9/20}K_{\le2}^{11/20}
 X^{11/20}Y^{3/20}.                                 \tag{4.3}
\]
Consequently, for $Q\in H^{-11/10}$ and all $\eta,\zeta>0$,
\[
\boxed{
\begin{split}
 |\langle Q,\Pi_K(a)\rangle|
 \le{}&\eta X+\zeta Y\\
 &+C K_0^{3/2}K_{\le2}^{11/6}
 \eta^{-11/6}\zeta^{-1/2}
 \|Q\|_{H^{-11/10}}^{10/3}.
\end{split}}                                        \tag{4.4}
\]
For spatially constant matrices, $K_{\le2}=K_0$ and the coefficient in the
remainder becomes $K_0^{10/3}$, exactly recovering R-118.

\paragraph{Proof.}
The $L^2$ estimate follows from the pointwise spectral bound, Holder, and the
same sharp-maximal centering used in R-118.  Differentiating the quadratic
output twice assigns at most two derivatives among $a$, the second copy of
$a$, and $K_r$.  The summed bilinear bounds (4.1), the algebra property of
the finite-cutoff smooth representative, and the R-118 square-function
argument give the second estimate in (4.2).  Real interpolation gives (4.3).
Duality against $Q$, followed by the same two-parameter Young optimization as
R-118, multiplies the interpolation exponents by $10/3$:
\[
 \frac9{20}\frac{10}3=\frac32,\qquad
 \frac{11}{20}\frac{10}3=\frac{11}6.
\]
This proves (4.4). \hfill$\square$

The order-two hypothesis cannot be deleted.  With constant $a$ and
$K_N(x)=\cos(Nx_1)\Id$, the pointwise operator norm stays bounded while the
$H^2$ norm of the quadratic output grows like $N^2$.  This is an abstract
method counterfixture, not a production counterexample.

\section*{5. Round four: stationary production low chaoses}

R-119 reduces zero/first-chaos cancellation to two aggregate identities.  If
$Y=b+A\xi+R$, with $R$ orthogonal to zero and first chaos, and $\delta\tau$
denotes the complete trace owner, define
\[
 D_0=\E\delta\tau-\E\|R\|^2,                        \tag{5.1}
\]
\[
 D_1=\E D\delta\tau
 -2\Contr(A,\E D^2R)-2\E[(DR)^*R].                 \tag{5.2}
\]
The low chaoses vanish exactly when $D_0=D_1=0$.

\subsection*{Proposition 5.1 (exact one-pair production checksum)}

For the actual first linear row, let
\[
 z=\frac\sigma2(\xi-i\eta),\qquad
 \xi,\eta\sim N(0,1),\qquad
 \lambda_1=4c_0=\frac6{125P}.
\]
The square-minus-trace diagnostic is
\[
 \cP=\frac{\lambda_1\sigma^4}{16}
 \left[(\xi^2+\eta^2)^2-4(\xi^2+\eta^2)\right].   \tag{5.3}
\]
Its exact Hermite expansion is
\[
 \cP_2=\frac{\lambda_1\sigma^4}{4}
 \{H_2(\xi)+H_2(\eta)\},                           \tag{5.4}
\]
\[
 \cP_4=\frac{\lambda_1\sigma^4}{16}
 \{H_4(\xi)+H_4(\eta)+2H_2(\xi)H_2(\eta)\}.      \tag{5.5}
\]
Therefore $\Pi_0\cP=\Pi_1\cP=0$.  Equivalently,
\[
 \E\tau=\E\|Y\|^2=\lambda_1\sigma^4,
 \qquad \E D\tau=2\E[(DY)^*Y]=0.                  \tag{5.6}
\]

\subsection*{Theorem 5.2 (stationary six-row A1 raw-current pass)}

Let $U$ be the complete fixed-cutoff stationary A1 Gaussian field with a
common entrywise-real-even scalar regulator.  For each spatial direction put
\[
 Y_i=C(U)\partial_iU,\qquad
 \tau_i=\Tr\bigl(C(U)^TC(U)\Gamma_i\bigr),          \tag{5.7}
\]
where $C$ contains all six rows in (2.3) and $\Gamma_i$ is the derivative
covariance.  Then the complete stationary packet satisfies both R-119
low-chaos identities:
\[
 \E\tau=\E\|Y\|^2,\qquad
 \E D\tau=2\E[(DY)^*Y]=0.                           \tag{5.8}
\]

\paragraph{Proof.}
Real-even stationarity gives
\[
 \E[U\otimes\partial_iU]=0.
\]
Because the pair is jointly Gaussian, $U$ and $\partial_iU$ are independent.
Conditioning on $U$ therefore proves
\[
 \E[\|Y_i\|^2\mid U]
 =\Tr\bigl(C(U)^TC(U)\Gamma_i\bigr)=\tau_i.        \tag{5.9}
\]
Every $q_A$ is even and every coefficient row is odd, so
$C(-U)=-C(U)$.  Under simultaneous sign reversal of the complete Gaussian
root, both $Y_i$ and $\tau_i$ are even.  Hence their zero/first-chaos
decompositions have $b=0$, $A=0$, $\E D\tau=0$, and
$\E[(DY)^*Y]=0$; the last two integrands are odd.  Summing over directions
and rows proves (5.8). \hfill$\square$

This result includes all six raw-current rows, including the rational rows.
It is stronger than the one-pair checksum, but it is not an owner-complete
trace/heat/low/R-063 forest theorem and is deliberately not stated for an
adapted triangular shift.

\section*{6. Round five: linear terminal Hessian}

\subsection*{Theorem 6.1 (exact linear-row Hessian)}

For every smooth cylindrical direction $z$,
\[
\boxed{
\begin{split}
 D^2V_L^{\rm ren}(u)[z,z]
 ={}&c_0\sum_A\|\nabla\mu_A\|_2^2
 -2c_0\sum_A\langle\Delta m_A,n_A\rangle\\

The first line is a positive square plus a fixed-selfadjoint centered

\section*{7. General current Hessian and rational Cartan survivor}

Let $\theta=f(u)\cdot\dd u$ be a one-form and set
\[
 j_i=\theta(\partial_i u),\quad \sigma=\theta(z),
 \quad \tau=D\theta[z](z),\quad \Omega=\dd\theta.
\]
The first variation has the invariant form
\[
 \dot j_i=\partial_i\sigma+\Omega(z,\partial_i u). \tag{7.1}
\]

\subsection*{Theorem 7.1 (current-Hessian identity)}

On the torus,
\[
\begin{split}
 D^2\frac12\sum_i\|j_i\|_2^2
 =\sum_i\int\{&|\partial_i\sigma+\Omega(z,\partial_i u)|^2
 -(\partial_i j_i)\tau\\
 &+j_i[(D_z\Omega)(z,\partial_i u)
       +\Omega(z,\partial_i z)]\}.
\end{split}                                                    \tag{7.2}
\]
The covariance-normal trace contributes exactly
\[
 -\sum_i\int\left\{(Df[z])^T\Gamma_iDf[z]
 +(D^2f[z,z])^T\Gamma_i f\right\}.                 \tag{7.3}
\]
Equations (7.2)--(7.3) retain all sign and factor information before any
analytic estimate.

For the rational row
\[
 \theta_A=\dd m_A-\alpha q_A\dd\rho,
\]
the curvature is
\[
\boxed{
 \Omega_A=\dd\theta_A
 =-\frac\alpha d\,\dd m_A\wedge\dd\rho.}           \tag{7.4}
\]
Thus the terminal Hessian contains the first-order Cartan block
\[
 \cC_A(z)=\sum_i\int j_{A,i}
 \left[(D_z\Omega_A)(z,\partial_i u)
       +\Omega_A(z,\partial_i z)\right].            \tag{7.5}
\]
It vanishes on special radial loci but is generically nonzero.  No
zeroth-order fixed-matrix theorem can silently replace (7.5).

\section*{8. Rational raw-covariance Hessian and fixed basis}

Define
\[

\[
 F_A=T_A-\alpha u\otimes g_A.                      \tag{8.3}

For a symmetric rough matrix $Q$, set

\subsection*{Theorem 8.1 (exact rational zeroth-order Hessian)}

The rational raw-$Q$ contribution satisfies
\[
 \frac12D^2B_R(u)[z,z]:Q
 =c_1\sum_Az^T\cK_A(Q)z,                            \tag{8.4}

where
\[
\boxed{
 \cK_A(Q)=4F_A^TQF_A-4\alpha\beta_AH_A
 -4\alpha(g_A\otimes w_A+w_A\otimes g_A).}         \tag{8.5}
\]
Each $\cK_A(Q)$ is selfadjoint.  Formula (8.5) is obtained by differentiating
$4c_1(T_Au)^TQ(T_Au)$ twice; no derived coefficient is pasted.

Every symmetric six-real matrix has the fixed expansion in
\[
 K_{pp}=e_pe_p^T,\qquad
 K_{pq}=\frac12(e_pe_q^T+e_qe_p^T),\quad p<q.       \tag{8.6}

There are $6\cdot7/2=21$ such matrices, and
\[

Hence the matrix direction of the zeroth-order rational Hessian is flattened

This does not finish the production multiplier estimate.  The scalar
