% evidence-anchor: lemma-3.1-four-moment-left-radau-majorant
\subsection*{Lemma 3.1 (left Radau majorant)}

Let $Z\ge0$ have finite fourth moment and a nondegenerate Stieltjes moment
matrix.  Define
\[
 D_0=m_1m_3-m_2^2,\qquad
 A_0={m_1m_4-m_2m_3\over D_0},\qquad
 B_0={m_2m_4-m_3^2\over D_0}.                              \tag{3.1}
\]
Let $0<r_1<r_2$ be the roots of $z^2-A_0z+B_0$.  There are unique positive
weights $w_0,w_1,w_2$ on $0,r_1,r_2$ matching $m_0,\ldots,m_4$; explicitly
\[
 w_1={m_1r_2-m_2\over r_1(r_2-r_1)},\qquad
 w_2={m_2-m_1r_1\over r_2(r_2-r_1)},\qquad
 w_0=1-w_1-w_2.                                           \tag{3.2}
\]
Then, for every $t\ge0$,
\[
 \boxed{\E e^{-tZ}\le w_0+w_1e^{-r_1t}+w_2e^{-r_2t}.}      \tag{3.3}
\]

\emph{Proof.}
The polynomial $z^2-A_0z+B_0$ is the monic degree-two orthogonal polynomial
for the positive measure $z\,d\Pr_Z$.  Hence its roots lie in $(0,\infty)$.
The Radau rule is exact through degree four, and each quadrature weight is
positive because it equals the expectation of the square of its cardinal
polynomial.  Let $H_t$ be the degree-four Hermite interpolant of $e^{-tz}$,
matching once at zero and twice at each $r_i$.  Its remainder is
\[
 e^{-tz}-H_t(z)={-t^5e^{-t\xi_z}\over5!}
 z(z-r_1)^2(z-r_2)^2\le0.                                \tag{3.4}
\]
Taking expectations and using degree-four exactness proves (3.3).
The bound is moment-sharp: the three-atom Radau law has the same four moments
and attains equality. \hfill$\square$

Applied to (1.11), Lemma 3.1 gives
\[
 \E e^{-tF_b}\le e^{\beta t}
 \{w_0+w_1e^{-r_1t}+w_2e^{-r_2t}\}.                       \tag{3.5}
\]

\subsection*{Lemma 4.1 (sufficient nonnegative skew)}

Assume
\[
 u\ge v>0,\qquad p\le\min(a,q).                           \tag{4.2}
\]
If $q\le a$, the third centered moment of the law on $0,v,v+u$ remains
nonnegative under every negative exponential tilt.  If $q>a$, put
\[
 k={u\over v},\qquad h={kq\over(k+1)a},                    \tag{4.3}
\]
\[
 F=\begin{cases}
 (q-a)/p,&h\le1,\\[2mm]
 {q\over p(k+1)}h^k,&h\ge1,
 \end{cases}                                               \tag{4.4}
\]
and
\[
 \Phi=(k-1)(k+2)(2k+1)
       +k^3{q-p\over q}+(k+1)^3{a-p\over q}-F.             \tag{4.5}
\]
Then $\Phi\ge0$ is sufficient for the same all-tilt conclusion.

\emph{Proof.}
Under the tilt, write
\[
 A=a,\qquad B=qe^{-vt},\qquad C=pe^{-(u+v)t},\qquad S=A+B+C.
\]
Direct expansion gives
\[
\begin{aligned}
S^3\mu_{3,t}={}&AB(A-B)v^3+AC(A-C)(u+v)^3+BC(B-C)u^3\\
&+ABC(u-v)(u+2v)(2u+v).
\end{aligned}                                               \tag{4.6}
\]
If $q\le a$, every term is nonnegative.  Otherwise the only adverse interval
has $B>A$.  On that interval, division by $ABC$ and (4.2) give
\[
{S^3\mu_{3,t}\over ABCv^3}\ge
-{B-A\over C}+k^3{q-p\over q}+(k+1)^3{a-p\over q}
 +(k-1)(k+2)(2k+1).                                      \tag{4.7}
\]
Indeed $B>A$ implies $e^{-vt}>a/q$, which yields
$(B-C)/A\ge(q-p)/q$; the other pair is bounded similarly.
With $s=e^{vt}\in[1,q/a]$,
\[
 {B-A\over C}={(q-as)s^k\over p}.                         \tag{4.8}
\]
Its derivative has the sign of $kq-(k+1)as$, so its exact maximum is (4.4).
Equations (4.5)--(4.8) prove the claim. \hfill$\square$

If $\Psi(t)$ is the logarithm of the right side of (3.5), then
\[
 \Psi''(t)=\Var_t(Z),\qquad \Psi'''(t)=-\mu_{3,t}(Z).      \tag{4.9}
\]
Thus Lemma 4.1 makes $\Psi''$ nonincreasing.
