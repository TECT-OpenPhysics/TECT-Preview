\section{Partial Wick is a diagnostic, not a replacement owner}

Let $W$ be a finite-dimensional isonormal Gaussian process.  For scalar
polynomials $F,G$ the full Wick product satisfies
\[
 \boxed{
 F\diamond G
 =\sum_{r\ge0}{(-1)^r\over r!}
   \langle D^rF,D^rG\rangle.}                         \tag{3.1}
\]
The sum terminates.  For Hilbert-valued polynomials the same formula applies
componentwise and is basis independent.

\emph{Proof.}
For the exponential vector
\[
 \mathcal E(h)=\exp\{W(h)-\|h\|^2/2\},               \tag{3.2}
\]
one has
\[
 \mathcal E(h)\diamond\mathcal E(k)=\mathcal E(h+k).
\]
On the other hand, $D^r\mathcal E(h)=h^{\otimes r}\mathcal E(h)$, so the
right side of (3.1) becomes
\[
 \mathcal E(h)\mathcal E(k)e^{-\langle h,k\rangle}
 =\mathcal E(h+k).                                    \tag{3.3}
\]
Differentiate in $h,k$ at zero to obtain every polynomial chaos. \hfill$\square$

If only root $j$ is Wick ordered and $F$ has root degree $N$, then for any
declared trace owner $\tau$,
\[
\boxed{
 {1\over2}(\|F\|^2-\tau)
 ={1\over2}\langle F\diamond_jF\rangle
 +{1\over2}(\|D_jF\|^2-\tau)
 +{1\over2}\sum_{r=2}^N{(-1)^{r+1}\over r!}
   \|D_j^rF\|^2.}                                    \tag{3.4}
\]
Conditionally on the other roots,
\[
 \E_j\langle F\diamond_jF\rangle=\|\E_jF\|^2.       \tag{3.5}
\]
For affine $F$ and $\tau=\|D_jF\|_{\HS}^2$, every remainder in (3.4)
vanishes.  This is the frozen R-107 situation.

\section{Exact quadratic recession classifier}

Let $X\sim N(0,\Id_n)$, let $Q:\R^n\to H$ be homogeneous quadratic, let
$A:\R^n\to H$ be linear, let $b\in H$, and let $\tau$ be a continuous scalar
homogeneous quadratic form.  Define
\[
 L(x)={1\over2}\{\|b+Ax+Q(x)\|^2-\tau(x)\},          \tag{5.1}
\]
\[
 U_q(x)={1\over2}\{|x|^2-q\tau(x)
                  +q\|b+Ax+Q(x)\|^2\}.              \tag{5.2}
\]
Then $\E e^{-qL(X)}$ is a constant times $\int e^{-U_q(x)}dx$.

Put $Z_Q=Q^{-1}(0)$.  For $v\in Z_Q\cap S^{n-1}$ define the parabolic tangent
image
\[
 \mathcal T_Q(v)=\left\{z:\ \exists r_j\to\infty,
 v_j\to v,\ |v_j|=1,\ r_jQ(v_j)\to z\right\}.        \tag{5.3}
\]

% evidence-anchor: theorem-5.1-exact-parabolic-tangent-recession
\subsection*{Theorem 5.1 (exact parabolic-tangent recession)}

With $\inf\varnothing=+\infty$,
\[
\boxed{
 \lambda_q:=\liminf_{|x|\to\infty}{2U_q(x)\over|x|^2}
 =\inf_{\substack{v\in Z_Q\cap S^{n-1}\\z\in\mathcal T_Q(v)}}
 \left\{1-q\tau(v)+q\|Av+z\|^2\right\}.}            \tag{5.4}
\]
Moreover:
\[
 \lambda_q>0\quad\Longrightarrow\quad \E e^{-qL(X)}<\infty, \tag{5.5}
\]
\[
 \lambda_q<0\quad\Longrightarrow\quad \E e^{-qL(X)}=\infty. \tag{5.6}
\]
The case $\lambda_q=0$ is not decided by quadratic recession.

\emph{Proof.}
For $x=rv$, $|v|=1$,
\[
 {2U_q(rv)\over r^2}
 =1-q\tau(v)+q\left\|Av+rQ(v)+{b\over r}\right\|^2. \tag{5.7}
\]
Along a finite recession sequence, compactness supplies a convergent
subsequence $v_j\to v$.  Finiteness forces $Q(v_j)\to0$ and boundedness of
$r_jQ(v_j)$, producing (5.3).  Conversely, every sequence in (5.3) inserted
in (5.7) realizes its displayed recession value.  This proves (5.4).  If
$Z_Q\cap S^{n-1}$ is empty, compactness gives $\inf_S\|Q\|>0$, and both sides
are $+\infty$.

If $\lambda_q>0$, then outside a ball
$U_q(x)\ge\lambda_q|x|^2/4$, proving (5.5).  If $\lambda_q<0$, select
$x_j$ with $U_q(x_j)\le-c|x_j|^2$.  Since $U_q$ is a polynomial of degree at
most four, $|\nabla U_q|\le C|x|^3$ on comparable-radius annuli.  A ball of
radius $\delta/|x_j|$ therefore retains $U_q\le-c|x_j|^2/2$.  Pass to a
disjoint subsequence.  Its integral is bounded below by
\[
 \sum_j C|x_j|^{-n}e^{c|x_j|^2/2}=\infty,             \tag{5.8}
\]
which proves (5.6). \hfill$\square$

The predictable baseline $b$ disappears only from strict quadratic
recession.  It can decide the equality case through lower-order terms.

\subsection*{Why the derivative range is not enough}

Always
\[
 \Ran DQ(v)\subseteq\mathcal T_Q(v).                  \tag{5.9}
\]
At a singular null, if $DQ(v)\zeta=0$, then second-order approaches can also
produce
\[
 DQ(v)\eta+Q(\zeta)\in\mathcal T_Q(v).                \tag{5.10}
\]
For example,
\[
 Q(x_1,x_2,x_3)=(x_1x_2,x_3^2),\qquad v=e_1,          \tag{5.11}
\]
has $\Ran DQ(v)=\operatorname{span}\{(1,0)\}$, while
\[
 v_r=(\sqrt{1-r^{-1}},0,r^{-1/2})
\]
gives $rQ(v_r)\to(0,1)$.  Thus replacing $\mathcal T_Q(v)$ by the derivative
range is false without an error bound.

A sufficient local hypothesis is the metric regularity estimate
\[
 \dist(w,Z_Q\cap S^{n-1})\le C\|Q(w)\|               \tag{5.12}
\]
near $v$, together with continuity of $DQ$.  Then
\[
 \mathcal T_Q(v)=\Ran DQ(v),                           \tag{5.13}
\]
and the inner infimum becomes
\[
 \dist\{-Av,\Ran DQ(v)\}^2
 =\dist\{Av,\Ran DQ(v)\}^2.                          \tag{5.14}
\]
Without linearity of the tangent image, the safe expression is
$\dist(-Av,\mathcal T_Q(v))^2$; its sign cannot be silently changed.

An unconditional but stronger sufficient condition is
\[
 \sup_{v\in Z_Q\cap S^{n-1}}q\tau(v)<1.              \tag{5.15}
\]

% evidence-anchor: proposition-5.2-equality-codimension-model
\subsection*{Proposition 5.2 (equality depends on codimension)}

Let $x=(t,y)\in\R\times\R^k$ and
\[
 Q(t,y)=ty,\qquad A=b=0,\qquad \tau(t,y)=t^2/q.       \tag{5.16}
\]
Then $\lambda_q=0$ and, up to a finite dimensional constant,
\[
 \E e^{-qL}=\int_\R(1+qt^2)^{-k/2}dt.                \tag{5.17}
\]
This diverges for $k=1$ and is finite for $k>1$.  Thus a zero recession
margin requires the critical-stratum dimension, normal rank, lower-order
terms, and singular strata; a sign check alone cannot close the endpoint.

% evidence-anchor: theorem-7.1-r082-exact-null-classification
\subsection*{Theorem 7.1 (pointwise and finite-Fourier null classification)}

\begin{enumerate}
\item if $u=0$, every tangent $(v,w)$ is null;
\item if $u\ne0$, then $v=i\theta u$ for real $\theta$ and

\end{enumerate}

$\Xi(\Psi,\partial_i\Psi)=0$ in every spatial direction.  Either

equivalently, the Newton polytope of a product is the Minkowski sum, and the
sum in (7.7) is a point only when each polytope is a point.  Apply the same

\section{Devil's-advocate review}

      \textbf{UPHELD AGAINST THAT ROUTE.}  Equations (3.6)--(3.8) exhibit the

      abstract endpoint has a sharp null boundary.

      \textbf{UPHELD AGAINST THAT ROUTE.}  Proposition 6.1 has mean zero,
      $H=128$, $K_W=1088$, and divergence already at $q=1/4$.

      \textbf{UPHELD AGAINST THAT ROUTE.}  Singular second-order approaches
      enlarge $\mathcal T_Q(v)$, as (5.11) shows.  The exact sign is (5.4).

\item \textbf{Objection: ``The derivative-range distance formula is always
      equivalent.''}  \textbf{UPHELD AGAINST THAT ROUTE.}  It requires the
      metric error bound (5.12).  Without a linear tangent image the safe
      distance is to $-Av$, not to $Av$.

      coercivity.''}  \textbf{UPHELD AGAINST THAT ROUTE.}  Real phase
      modulations lie in the linearized kernel, and near-null parabolic
      sequences are not controlled by exact-zero rigidity.

      Componentwise parity fails after that coordinate deletion.  The face
      proves pointwise noncoercivity only; the full stationary root is centered.

      derivative even at realized $u_2=0$.

      \textbf{UPHELD AGAINST THAT ROUTE.}  The exact fixed-cutoff minimum is
      (7.20), but $K_j^{\max}$ is not cutoff uniform and the terminal sextic
      is a one-use global owner.
