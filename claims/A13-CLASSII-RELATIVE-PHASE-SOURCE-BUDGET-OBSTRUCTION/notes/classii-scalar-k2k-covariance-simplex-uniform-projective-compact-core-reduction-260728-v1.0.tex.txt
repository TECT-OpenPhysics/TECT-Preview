\section{Closed covariance-simplex normal form}

Let $\gamma=v+4w>0$ and set
\[
 c={v\over\gamma},\qquad s={4w\over\gamma},\qquad
 b={A^2\over\gamma},\qquad \tau=q\gamma^2,
 \qquad x=b\tau.                                      \tag{1.1}
\]
Thus $c,s\ge0$ and $c+s=1$.  Let $T,U$ be independent unit exponentials and
put
\[
 \rho={cT\over2},\qquad \sigma={sU\over8}.             \tag{1.2}
\]
At $c=0$ or $s=0$ the zero-weight exponential is a harmless dummy variable.
The relative phase $\phi$ is uniform.  With
\[
 L=\rho+4\sigma-{1\over2},\qquad
 Q=\rho^2+10\rho\sigma+4\sigma^2-\rho-\sigma,
 \qquad B=6\rho\sqrt\sigma\cos\phi,                  \tag{1.3}
\]
the exact R-110 packet and covariance square become
\[
 \boxed{F:={P\over\gamma^2}=bL+\sqrt b\,B+Q,}          \tag{1.4}
\]
\[
 \boxed{K:={H\over\gamma^4}=b^2K_2+bK_1+K_0,}         \tag{1.5}
\]
where
\[
 K_2={c^2+s^2\over2},                                   \tag{1.6}
\]
\[
 K_1=c^3+{5\over2}c^2s+cs^2+{s^3\over4},              \tag{1.7}
\]
\[
 K_0={5\over4}c^4+{1\over4}c^3s+{25\over16}c^2s^2
       +{1\over4}cs^3+{5\over64}s^4.                   \tag{1.8}
\]
The packet is centered.  The exact local margin is
\[
 \boxed{\Delta(c,s):=K-2\Var(F)
 ={c^4\over4}+{c^2s^2\over4}+{s^4\over64}.}            \tag{1.9}
\]
It has the elementary uniform lower bound
\[
 64\Delta=(4c^2+s^2)^2+8c^2s^2,\qquad
 4c^2+s^2={4\over5}+5\left(c-{1\over5}\right)^2,
 \qquad \Delta\ge {1\over100}.                         \tag{1.10}
\]

The phase-integrated Laplace transform is best written directly in the
compact variables:
\[
 \boxed{
 M(c,x,\tau)=\E\!\left[e^{-xL-\tau Q}
 I_0\!\left(6\sqrt{x\tau}\,\rho\sqrt\sigma\right)\right].}
                                                                  \tag{1.11}
\]
The desired scalar theorem is
\[
 \boxed{\log M(c,x,\tau)\le
 {x^2K_2+x\tau K_1+\tau^2K_0\over4}.}                  \tag{1.12}
\]
For finite $b$ one has $x=b\tau$.  Allowing $\tau=0$ and $x>0$ adds the
large-amplitude projective boundary and makes the parameter space compact.

\section{Exact global exterior and a compact interval core}

% evidence-anchor: theorem-2.1-global-floor-compact-semialgebraic-core
\subsection*{Theorem 2.1 (global floor and compact semialgebraic reduction)}

For every $b,\rho,\sigma\ge0$,
\[
\boxed{
\begin{aligned}
 F_-={}&\left(\rho-{1\over2}\right)^2
 +4\left(\sigma-{1\over8}\right)^2\\
 &+\rho\left[(\sqrt b-3\sqrt\sigma)^2+\sigma\right]
 +4b\sigma-{b\over2}-{5\over16}.
\end{aligned}}                                           \tag{2.2}
\]
Hence
\[
 F\ge-{b\over2}-{5\over16}.                            \tag{2.3}
\]
On the covariance side,
\[
 K_2={1+(c-s)^2\over4},                                  \tag{2.4}
\]
\[
 K_1={1\over4}+{c(1+5c-3c^2)\over4},                   \tag{2.5}
\]
\[
 K_0={1\over16}+{s^4+76c^2s^2+76c^4\over64}.           \tag{2.6}
\]
Every remainder is nonnegative on $c+s=1$, so
\[
 \boxed{K\ge{(2b+1)^2\over16}.}                         \tag{2.7}
\]
The floor proves (1.12) whenever
\[
 \tau\ge R(b):={32b+20\over(2b+1)^2}.                  \tag{2.8}
\]
Indeed $\log M\le\tau(b/2+5/16)$, and the right side of (1.12) is at
least $\tau^2(2b+1)^2/64$.

Moreover,
\[
 20(2b+1)^2-(32b+20)=16b(5b+3)\ge0,                    \tag{2.9}
\]
\[
 8(2b+1)^2-b(32b+20)=4(3b+2)>0.                         \tag{2.10}
\]
Thus only $0\le\tau\le20$ and $0\le x=b\tau\le8$ remain.  Eliminating
$b=x/\tau$ gives the exact closed residual condition
\[
 \boxed{(2x+\tau)^2\le32x+20\tau.}                     \tag{2.11}
\]
This proves the claimed compactification. \hfill$\square$

\section{Uniform projective boundary}

For $b>0$, let $\delta=b^{-1}=\tau/x$ and keep $x$ fixed.  Then
\[
 \tau F=x(L+\sqrt\delta\,B+\delta Q).                  \tag{3.1}
\]
Under the leading tilt, $T$ and $U$ remain independent exponentials with
rates
\[
 \alpha=1+{cx\over2},\qquad \beta=1+{sx\over2}.         \tag{3.2}
\]
The limiting transform and gap are
\[
 M_\infty={e^{x/2}\over(1+cx/2)(1+sx/2)},               \tag{3.3}
\]
\[
 \boxed{D_0=d(cx/2)+d(sx/2),\qquad
 d(u)={u^2\over2}-u+\log(1+u).}                         \tag{3.4}
\]
Since $d'(u)=u^2/(1+u)$, $D_0>0$ for $x>0$.  On $0\le x\le8$,
\[
 d(u)\ge{u^3\over15}\quad(0\le u\le4),\qquad
 c^3+s^3\ge{1\over4},
\]
and hence
\[
 \boxed{D_0\ge{x^3\over480}.}                           \tag{3.5}
\]

% evidence-anchor: theorem-3.1-uniform-projective-expansion
\subsection*{Theorem 3.1 (uniform expansion through the second correction)}

Uniformly for $c+s=1$ and $0\le x\le8$,
\[
 \boxed{
 G_b:={\tau^2K\over4}-\log M
 =D_0+{D_1\over b}+{D_2\over b^2}
   +O\!\left({x^3\over b^3}\right).}                   \tag{3.6}
\]
The first coefficient is
\[
 \boxed{D_1={x^3N_1(c,s,x)\over
 16(2+cx)^2(2+sx)^2}\ge0,}                              \tag{3.7}
\]
where
\begin{equation}
\begin{aligned}
N_1={}&4c^5s^2x^3+16c^5sx^2+16c^5x
+10c^4s^3x^3+56c^4s^2x^2+104c^4sx+64c^4\\
&+4c^3s^4x^3+56c^3s^3x^2+172c^3s^2x+152c^3s\\
&+c^2s^5x^3+20c^2s^4x^2+88c^2s^3x+112c^2s^2\\
&+4cs^5x^2+32cs^4x+32cs^3+4s^5x+16s^4.
\end{aligned}                                                \tag{3.8}
\end{equation}
All coefficients are positive.

The second coefficient satisfies the stronger exact bound
\[
 \boxed{D_2\ge{x^2\Delta(c,s)\over4}\ge{x^2\over400}.} \tag{3.9}
\]

\emph{Uniform remainder.}
Write $\varepsilon=b^{-1/2}$.  For $0<\theta\le1$ and $b\ge10$, put
$b_\theta=(\theta\varepsilon)^{-2}\ge10$.  The intermediate exponent
$L+\theta\varepsilon B+\theta^2\varepsilon^2Q$ is $F_{b_\theta}/b_\theta$.
The sharper phase-minimum identity
\[
 b_\theta-1+10\sigma-6\sqrt{b_\theta\sigma}
 =10\left(\sqrt\sigma-{3\sqrt{b_\theta}\over10}\right)^2
   +{b_\theta\over10}-1\ge0
\]
gives $F_{b_\theta}\ge-b_\theta/2$ and hence the common dominator
$e^{x/2}e^{-T-U}$; the endpoint $\theta=0$ follows by continuity.  The sixth
$\varepsilon$ derivative is a Bell polynomial
in $-xB-2x\varepsilon Q$ and $-2xQ$.  Every monomial has $x$-degree at least
three, and $B,Q$ have simplex-uniform polynomial coefficients.  Thus the
integrated sixth derivative is bounded by
\[
 Cx^3(1+T+U)^Ne^{-T-U}                                  \tag{3.10}
\]
for one finite constant on $x\le8$.  Phase symmetry removes odd powers.

If
\[
 A_1=\E_x\left[{x^2B^2\over2}-xQ\right],               \tag{3.11}
\]
\[
 A_2=\E_x\left[{x^4B^4\over24}-{x^3B^2Q\over2}
                         +{x^2Q^2\over2}\right],        \tag{3.12}
\]
then
\[
 D_1={x^2K_1\over4}-A_1,\qquad
 D_2={x^2K_0\over4}-\left(A_2-{A_1^2\over2}\right).   \tag{3.13}
\]
The centered identity $\E_0Q=0$ and smooth tilted moments give
$A_1=O(x^2)$ and $A_2=O(x^2)$.  Thus the logarithmic sixth-order combinations
$A_1A_2$ and $A_1^3$ are respectively $O(x^4)$ and $O(x^6)$, hence
$O_8(x^3)$.  Together with (3.10), this proves that passing from the raw
transform to its logarithm preserves the factored $O(x^3b^{-3})$ remainder.

Exact exponential moments give (3.7).  After $s=1-c$, the exact positive
denominators for $D_2/x^2$ and
$(D_2-x^2\Delta/4)/x^3$ are respectively
\[
 256(2+cx)^4(2+(1-c)x)^4,
 \qquad 64(2+cx)^4(2+(1-c)x)^4.
\]
They are strictly positive on $0\le c\le1$, $x\ge0$.  Clear these
denominators and express both numerators in the degree-twelve Bernstein basis
in $c$.  Each of the thirteen Bernstein coefficients is coefficientwise
nonnegative in $x$.  The minimum power coefficients for the margin reserve are
\[
 16,{8\over3},{4\over11},{2\over55},{1\over495},{1\over198},
 {1\over44},{1\over198},{16\over495},{32\over55},
 {64\over11},{128\over3},256.                            \tag{3.14}
\]
This is a finite exact polynomial certificate.  Equations (3.10)--(3.14)
prove (3.6)--(3.9). \hfill$\square$
