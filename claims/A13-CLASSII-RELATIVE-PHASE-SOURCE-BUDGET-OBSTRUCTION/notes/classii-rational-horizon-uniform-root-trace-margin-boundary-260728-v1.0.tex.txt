\section{A full-horizon tail theorem for rational maps}

Let $F:\R^n\to\R^m$ be locally Lipschitz with at most quadratic growth, let
$\tau:\R^n\to\R$ be continuous with at most quadratic growth, and assume the
uniform spherical limits, in the explicit sense
\[
 \sup_{|v|=1}\left\|{F(rv)\over r^2}-Q_\infty(v)\right\|
 +\sup_{|v|=1}\left|{\tau(rv)\over r^2}-\tau_\infty(v)\right|
 \longrightarrow0.                                \tag{2.1}
\]
The maps on the right are continuous.  Put
\[
 \cZ_\infty=\{v\in S^{n-1}:Q_\infty(v)=0\}.       \tag{2.2}
\]
For $v\in\cZ_\infty$, define the full horizon tangent image
\[
 \cT_F(v)=\left\{z:\begin{array}{l}
  r_j\to\infty,\ v_j\to v,\ |v_j|=1,\\
  F(r_jv_j)/r_j\to z
 \end{array}\right\}.                              \tag{2.3}
\]
It can be empty and need not equal the range of a first derivative.  Given
$b\in\R^m$, a linear map $A:\R^n\to\R^m$, and $q>0$, set
\[
 U_q(x)={1\over2}\left\{|x|^2-q\tau(x)
          +q\|b+Ax+F(x)\|^2\right\}.               \tag{2.4}
\]

% evidence-anchor: theorem-2.1-full-rational-horizon-classifier
\subsection*{Theorem 2.1 (full rational-horizon classifier)}

With the convention $\inf\varnothing=+\infty$,
\[
 \boxed{
 \liminf_{|x|\to\infty}{2U_q(x)\over|x|^2}
 =\lambda_q:=
 \inf_{\substack{v\in\cZ_\infty\\z\in\cT_F(v)}}
 \{1-q\tau_\infty(v)+q\|Av+z\|^2\}.}             \tag{2.5}
\]
If $\lambda_q>0$, then $e^{-U_q}$ is integrable.  If $\lambda_q<0$ and
there are $C,M,R_0<\infty$ such that
\[
 \operatorname{Lip}\bigl(U_q;\{R/2\le|x|\le2R\}\bigr)
 \le CR^M\qquad(R\ge R_0),                          \tag{2.6}
\]
then $e^{-U_q}$ is not integrable.  The equality case $\lambda_q=0$ is not
decided by recession alone.

\emph{Proof.}
Write $x_j=r_jv_j$.  If $2U_q(x_j)/r_j^2$ is bounded above, positivity of the
last square in (2.4) makes $F(r_jv_j)/r_j$ bounded.  After a subsequence,
$v_j\to v$ and $F(r_jv_j)/r_j\to z$.  Uniformity in (2.1) then forces
$Q_\infty(v)=0$, while $b/r_j\to0$, and the subsequential limit is the
quantity in braces in (2.5).  Conversely, every $z\in\cT_F(v)$ is realized
by its defining sequence and gives the reverse inequality.  This proves
(2.5).

If $\lambda_q>0$, (2.5) gives $U_q(x)\ge c|x|^2$ outside a ball.  If
$\lambda_q<0$, a realizing sequence has $U_q(x_j)\le-c|x_j|^2$.  A
polynomial annular Lipschitz bound preserves this inequality, with a smaller
$c$, on balls of radius comparable to
$\min\{r_j,r_j^{2-M}\}$.  Passing to a separated subsequence, their volumes
are only polynomially small while their exponential masses grow like
$e^{cr_j^2}$, so the sum diverges.
\hfill$\square$

This theorem is deliberately formulated with $\cT_F(v)$ rather than
$\operatorname{Ran}DF(v)$.  R-116 already showed that singular parabolic
approaches can create horizon tangents outside the derivative range.

\section{The R-082 positive-floor horizon}

\[

\]
and put
\[

\]
For $e>0$, the R-082 four-coordinate map is
\[
 \Xi_e(z,y)=\begin{pmatrix}
 \sqrt{c_0}\,dr\\
 \sqrt{c_1}\{dr-\alpha r(\rho+e)^{-1}d\rho\}\\
 2\sqrt{c_0+c_1}\Re h\\
 2\sqrt{c_0+c_1}\Im h
 \end{pmatrix},                                    \tag{3.3}
\]
where
\[
 c_0={3\over250P},\qquad c_1={243\over8000P},
 \qquad\alpha={5\over9}.                           \tag{3.4}
\]
At $e=0$, the ratio $\beta=r/\rho$ has no continuous value at $\rho=0$.
Nevertheless the product
$g_\beta=((1-\alpha\beta)u,-\alpha\beta\chi)$ and the complete map $\Xi_0$

% evidence-anchor: proposition-3.1-uniform-floor-horizon
\subsection*{Proposition 3.1 (uniform positive-floor horizon)}

For $t>0$,
\[
 \Xi_e(tz,ty)=t^2\Xi_{e/t^2}(z,y),                 \tag{3.5}
\]
and, uniformly in $z$,
\[
 \boxed{
 \|t^{-2}\Xi_e(tz,ty)-\Xi_0(z,y)\|
 \le\sqrt{c_1}\,\alpha{\sqrt e\over t}|y|.}       \tag{3.6}
\]

\emph{Proof.}
Only the second row changes.  With $\delta=e/t^2$, its absolute difference
is at most
\[
 \sqrt{c_1}\alpha{r\delta\over\rho(\rho+\delta)}
 |d\rho|
 \le2\sqrt{c_1}\alpha{\delta\sqrt\rho\over
          \rho+\delta}|y|.                          \tag{3.7}
\]
The square of the normalized scalar factor is
\[
 {1\over\delta}\left({2\delta\sqrt\rho\over
 \rho+\delta}\right)^2={4x\over(1+x)^2}\le1,
 \qquad x={\rho\over\delta},                       \tag{3.8}
\]
because its defect from one is $(x-1)^2/(x+1)^2$.
\hfill$\square$

The bound is uniform in $z$ for the displayed tangent $y$.  After inserting
a derivative $y=Dz$, its spherical $O(t^{-1})$ constant can depend on the
finite cutoff.  Proposition 3.1 proves the horizon separately at every
cutoff; it is not a cutoff-uniform horizon-regularity estimate.

For fixed target heat $R$ independent of the fresh root, (3.6) and the
linear frame bound below give dominated convergence:
\[
 \lim_{t\to\infty}t^{-2}P_\Sigma B_e(tz)=B_\infty(z).
                                                               \tag{3.9}
\]
This is a recession statement.  The raw finite-amplitude estimate
$P_\Sigma B_e(z)\lesssim|z|^2$ is false at $z=0$ whenever the heat has
positive variance.

\section{Sharp homogeneous frame trace}

Let $Y_i$ be a proper circular complex Gaussian derivative with covariance
$D_i=\E[Y_iY_i^*]$, and write $d_i=\|D_i\|_{\rm op}$.  Put
$\beta=r/\rho$ and
\[
 g_\beta=((1-\alpha\beta)u,-\alpha\beta\chi).       \tag{4.1}
\]
Then
\[
 {|g_\beta|^2\over\rho}
 =f(\beta):=\beta-\alpha(2-\alpha)\beta^2
 =\beta-{65\over81}\beta^2.                       \tag{4.2}
\]
The separate rational row has maximum $81/260$, but optimizing all four
rows jointly is sharper.

% evidence-anchor: theorem-4.1-sharp-homogeneous-frame-trace
\subsection*{Theorem 4.1 (sharp homogeneous frame trace constant)}

For every $z$ and each spatial direction,
\[
 \E\|\Xi_0(z,Y_i)\|^2
 \le d_i\rho K(\beta),                              \tag{4.3}
\]
where
\[
 K(\beta)=6(c_0+c_1)\beta
           -2c_1{65\over81}\beta^2.                \tag{4.4}
\]
The function is increasing on $[0,1]$, and hence
\[
 \boxed{
 \E\|\Xi_0(z,Y_i)\|^2
 \le\kappa_*d_i\rho,
 \qquad \kappa_*={411\over2000P}.}                 \tag{4.5}
\]
The constant is attained when $\chi=0$ and $D_i$ is scalar on the active
doublet.

\emph{Proof.}
Proper complex covariance gives
\[
 \E(dr)^2=2u^*D_iu\le2d_ir,\qquad
 \E\{2\Re(g_\beta^*Y_i)\}^2\le2d_i|g_\beta|^2.
                                                               \tag{4.6}
\]
Also
\[
 \E|u_1(Y_i)_2-u_2(Y_i)_1|^2\le d_ir.              \tag{4.7}
\]
After multiplying by the four row weights, the sum is (4.3)--(4.4).
Moreover
\[
 K'(1)=6(c_0+c_1)-4c_1{65\over81}>0,               \tag{4.8}
\]
and $K'$ decreases linearly, so $K$ is increasing.  Finally
\[
 K(1)=6c_0+\left(4+{32\over81}\right)c_1
 ={411\over2000P}.                                  \tag{4.9}
\]
The covariance convention is
$\Gamma_\R=\operatorname{realify}(D_i)/2$; (4.6) already contains this
factor.  There is no additional Pauli, complex, or volume multiplier.
\hfill$\square$
