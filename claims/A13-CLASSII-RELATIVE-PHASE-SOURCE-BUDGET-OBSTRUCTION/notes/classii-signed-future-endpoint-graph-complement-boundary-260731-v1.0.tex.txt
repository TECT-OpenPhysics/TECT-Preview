\section*{3. Signed future terminal-minus-prefix theorem}

% evidence-anchor: theorem-3.1-signed-future-terminal-prefix
\subsection*{Theorem 3.1 (endpoint-first future telescope)}

Under the hypotheses of Section 2,
\[
 \boxed{\omega_{e,r,m}=K_{r,m}^{e}-K_{r,m}^{e-1}.}             \tag{3.1}
\]
Consequently
\[
 \boxed{
 \sum_{e>n(r)}\omega_{e,r,m}
   =K_{r,m}^{N}-K_{r,m}^{n(r)}.}                              \tag{3.2}
\]
For a collar $C\ge5$ and reveal weight
\[
 w_{r,m}^{(\gamma,C)}
 =2^{2\gamma(m-r-C)}\mathbf 1_{\{m\ge r+C\}},               \tag{3.3}
\]
the complete weighted future branch is therefore
\[
 \boxed{
 \mathfrak I_{\rm fut}^{(\gamma,C)}
 =\sum_{r>\ell}\sum_{m\ge r+C}
 w_{r,m}^{(\gamma,C)}
 \bigl(K_{r,m}^{N}-K_{r,m}^{n(r)}\bigr).}                    \tag{3.4}
\]
No physical insertion label $k$ occurs in (3.4), and hence no factor
$2^{2\gamma(k-r)}$ occurs.

\paragraph{Proof.}
For $x=Y_{r,m}^{e-1}$ and
$y=Y_{r,m}^{e}-Y_{r,m}^{e-1}$, polarization gives
\[
 \Rea\ip{x}{y}+{1\over2}\|y\|^2
 ={1\over2}(\|x+y\|^2-\|x\|^2).                      \tag{3.5}
\]
Subtract the matching primitive-trace increment. This is exactly (3.1).
The finite sum over every $e>n(r)$ cancels each intermediate endpoint and
proves (3.2). The weight (3.3) depends only on the fixed pair $(r,m)$, so it
may multiply the already telescoped identity. Summing in $(r,m)$ proves
(3.4). \hfill$\square$

Equation (3.4) is invariant under a representation-preserving subdivision
which preserves the filtration, reveal order, admissible terminal and prefix
controls, and output resolution. The individual $\omega_{e,r,m}$ need not be
invariant. A new root, a changed reveal order, or an enlarged admissible
source space is a different chart and is not covered by this assertion.

\section*{6. Why a wedge-only telescope is impossible}

Let $k_e$ be the physical grade of insertion $e$ and mask only the
reveal-FAR/insertion-near wedge by
\[
 \chi_e(r,m)=\mathbf 1_{\{r+C\le m<k_e+C\}}.                 \tag{6.1}
\]
For any endpoint sequence $K_e$ finite summation by parts gives
\[
\begin{split}
 \sum_{e=a}^{N}\chi_e(K_e-K_{e-1})
 ={}&\chi_NK_N-\chi_aK_{a-1}\\
 &+\sum_{e=a}^{N-1}(\chi_e-\chi_{e+1})K_e.
\end{split}                                             \tag{6.2}
\]
The last sum vanishes for the complete mask $\chi_e\equiv1$, but not for a
moving wedge. If $K_e$ alternates $0,1,0,1,\ldots,0$ and the mask selects
only rising increments, the terminal difference is zero while the masked sum
grows linearly with the number of rises. This establishes
\texttt{NG-2026-07-31-A13-WEDGE-ONLY-FUTURE-TELESCOPE}. It is a finite
algebraic method no-go, not a production-action counterexample.

The same obstruction appears as a cross-owner. With trace zero and
$y_F=u$, $y_N=-u$,
\[
 {1\over2}\|y_F+y_N\|^2=0,                              \tag{6.3}
\]
while the positive far square is $\|u\|^2/2$ and the cross-retaining near
owner is $-\|u\|^2/2$. Spending the far square separately consumes a
cancellation which the near owner still needs. A legal positive split would
require a new Bessel, Loewner, or causal Schur theorem for the complete
production columns.

\section*{8. Exact robust shifted-Douglas complement}

Let $X,Y,L$ be finite-dimensional real Hilbert spaces. Let
\[
 \mathcal M(T)=

 -(A_0+T)^*/2&F&-C\\

 \end{pmatrix},                                      \tag{8.1}
\]
where $T:Y\to X$ is the tail response and $A_0$ contains every balanced,
near, and future-wedge cross which is not in that tail. Put
\[
 K=\binom BC,\qquad
 M_{2,0}=\begin{pmatrix}E&-A_0/2\\-A_0^*/2&F\end{pmatrix}.  \tag{8.2}
\]
Fix a desired gap $\mu\ge0$, set $D_\mu=D-\mu\Id$, and assume
\[
 D_\mu\succeq0,\qquad \Ran K^*\subseteq\Ran D_\mu^{1/2}.    \tag{8.3}
\]
Let $D_\mu^{1/2}X_\mu=K^*$ be the Douglas reduced solution and define
\[
 S_\mu=M_{2,0}-\mu\Id-X_\mu^*X_\mu.                       \tag{8.4}
\]

% evidence-anchor: theorem-8.1-robust-complement
\subsection*{Theorem 8.1 (robust complement criterion)}

For $\tau\ge0$, the following are equivalent:
\[
 \mathcal M(T)\succeq\mu\Id
 \quad\hbox{for every }T:Y\to X\hbox{ with }\|T\|\le\tau, \tag{8.5}
\]
and
\[
 \boxed{
 \ip{(x,y)}{S_\mu(x,y)}
 \ge\tau\|x\|\|y\|
 \quad\hbox{for every }x\in X,\ y\in Y.}                  \tag{8.6}
\]

\paragraph{Proof.}
Shift (8.1) by $\mu\Id$ and apply the finite-dimensional Douglas/Schur
criterion to the low block. The remaining tail contribution to the reduced
quadratic form is $-\Rea\ip{x}{Ty}$. For every $\|T\|\le\tau$,
\[
 |\Rea\ip{x}{Ty}|\le\tau\|x\|\|y\|,                       \tag{8.7}
\]
so (8.6) is sufficient. Conversely, for fixed nonzero $x,y$, the rank-one
map sending $y/\|y\|$ to $\tau x/\|x\|$ attains the right side of (8.7).
Thus robustness for every such $T$ implies (8.6). \hfill$\square$

For one known tail $T_C$, the still weaker exact condition is simply the
positivity of (8.4) after subtracting its symmetrized tail block. The scalar
R-133/R-138 headroom follows as a conservative corollary. If
\[
 E\succeq e\Id,\quad F\succeq f\Id,\quad D\succeq d\Id,
 \quad\|K\|\le k,\quad\|A_0\|\le a_0,                     \tag{8.8}
\]
then, for the zero-gap case $\mu=0$, with $\sigma=k^2/d$, the condition
\[
 \boxed{
 a_0+\tau<2\sqrt{(e-\sigma)(f-\sigma)}}                    \tag{8.9}
\]
is sufficient when $e,f>\sigma$. It is not the weakest condition because it
replaces the reduced low operator by the scalar bound $\sigma\Id$.
For a prescribed gap $\mu>0$, the corresponding sufficient conditions are
\[
 d>\mu,\qquad
 \sigma_\mu={k^2\over d-\mu},\qquad
 e-\mu>\sigma_\mu,\quad f-\mu>\sigma_\mu,                  \tag{8.10}
\]
and
\[
 \boxed{
 a_0+\tau<
 2\sqrt{(e-\mu-\sigma_\mu)(f-\mu-\sigma_\mu)}.}           \tag{8.11}
\]

\section*{9. Tail-only and low-only strict-gap no-go}

\paragraph{Proof.}
