\section*{3. Exact antipodal field and current syntheses}
\hypertarget{section-3-synthesis}{ }

Let
\[

 \qquad U(\theta)=\cos\theta\,\Id_6+\sin\theta\,J.  \tag{3.1}
\]
Because $\Gamma(p)$ has the duplicated form (2.4),
\[
 [J,\Gamma(p)]=0,
 \qquad U(\theta)\Gamma(p)U(\theta)^*=\Gamma(p).    \tag{3.2}
\]
Reveal the two signs of one antipodal pair as a single Gaussian block
$g=(g_+,g_-)\sim N(0,\Id_{12})$.  With
$\theta_x=p\cdot x$, define
\[
 S_x(x)={1\over\sqrt V}
 \begin{bmatrix}
 U(\theta_x)G_p&U(-\theta_x)G_p
 \end{bmatrix},                                      \tag{3.3}
\]
\[
 S_{v,i}(x)={p_i\over\sqrt V}
 \begin{bmatrix}
 JU(\theta_x)G_p&-JU(-\theta_x)G_p
 \end{bmatrix}.                                      \tag{3.4}
\]
These are the A1 Fourier field and the physical derivative
$\partial_i\leftrightarrow ip_i$ in six-real coordinates.

\subsection*{Theorem 3.1 (coincident cancellation and nonlocal kernel)}

For every $x$,
\[
 S_x(x)S_x(x)^*={2\over V}\Gamma(p),\qquad
 S_{v,i}(x)S_{v,i}(x)^*={2p_i^2\over V}\Gamma(p),   \tag{3.5}
\]
and
\[
 \boxed{K_i(x,x)=S_x(x)S_{v,i}(x)^*=0.}             \tag{3.6}
\]
At two distinct points,
\[
 \boxed{
 K_i(x,y)={2p_i\over V}\Gamma(p)
            \sin\{p\cdot(x-y)\}.}                  \tag{3.7}
\]
Consequently $K_i(y,x)=-K_i(x,y)$ and the kernel is not the zero
operator.  For the final pair $p=q=2k$, (3.7) is
\[
 K_i(x,y)={4k_i\over V}\Gamma(2k)
            \sin\{2k\cdot(x-y)\}.                  \tag{3.8}
\]

\textit{Proof.}
Equation (3.2) gives each marginal immediately.  In the cross term the
$+p$ contribution is $-p_i\Gamma(p)J/V$ and the $-p$ contribution is
$+p_i\Gamma(p)J/V$, proving (3.6).  At different points the same two terms
combine as
\[
 {p_i\over V}\{U(\theta_y-\theta_x)
                 -U(\theta_x-\theta_y)\}\Gamma(p)J,
\]
which reduces to (3.7) because $J^2=-\Id_6$. \hfill$\square$

In the cosine/sine coefficient basis the same calculation reads
\[
 S_x^{\rm coef}=\begin{pmatrix}G_p&0\\0&G_p\end{pmatrix},
\qquad
 S_{v,i}^{\rm coef}=p_i
 \begin{pmatrix}0&G_p\\-G_p&0\end{pmatrix},         \tag{3.9}
\]
so
\[
 K_i^{\rm coef}=p_i
 \begin{pmatrix}0&-\Gamma(p)\\\Gamma(p)&0\end{pmatrix}
 \ne0.                                                \tag{3.10}
\]
Thus ``$K=0$'' in (3.6) is a coincident local statement, not a projected or
coefficient-space statement.
