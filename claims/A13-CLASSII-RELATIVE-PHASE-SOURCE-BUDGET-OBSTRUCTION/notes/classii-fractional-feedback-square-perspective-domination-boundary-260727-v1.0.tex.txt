\section{The moving-prefix square defect}

Let the terminal coefficient be held fixed in the following Hilbert-current
coordinates:
\[
\begin{aligned}
 \rho_*&=Q_{\rm II}^{1/2}M_*^T D(A^*-A^0),\\
 \rho_j^-&=Q_{\rm II}^{1/2}M_*^T D(A^{(j)}-A^0),\\
 \rho_j^+&=Q_{\rm II}^{1/2}M_*^T D(A^*-A^{(j)}).
\end{aligned}                                                \tag{3.1}
\]
Thus $\rho_* =\rho_j^-+\rho_j^+$ for every $j$.  Put
\[
 \zeta_j=\delta_j^{\rm D}\rho_j^- ,\qquad
 y_j=\delta_j^{\rm D}\rho_j^+ .                       \tag{3.2}
\]
The symbol $y_j$ agrees with the future derivative-feedback coordinate in
the fixed-terminal lift; it is not a spatial Gaussian derivative.

% evidence-anchor: theorem-3.1-moving-prefix-square-defect
\subsection*{Theorem 3.1 (exact moving-prefix square defect)}

For every square-integrable terminal current and every $0<\theta<1$,
\[
\boxed{
\begin{aligned}
 &{\theta\over2}\mathbb E\|\rho_*\|^2
 -{\theta\over2}\sum_j\mathbb E\|y_j\|^2\\
 &\quad={\theta\over2}\mathbb E\|P_\ell\rho_*\|^2
 +{\theta\over2}\sum_j\mathbb E\|\zeta_j\|^2
 +\theta\sum_j\mathbb E\langle\zeta_j,y_j\rangle .
\end{aligned}}                                                \tag{3.3}
\]

\emph{Proof.}
Applying $\delta_j^{\rm D}$ to $\rho_* =\rho_j^-+\rho_j^+$ gives
\[
 \delta_j^{\rm D}\rho_*=\zeta_j+y_j.                 \tag{3.4}
\]
Doob Pythagoras for the fixed terminal random variable $\rho_*$ gives
\[
 \mathbb E\|\rho_*\|^2
 =\mathbb E\|P_\ell\rho_*\|^2
 +\sum_j\mathbb E\|\zeta_j+y_j\|^2.                 \tag{3.5}
\]
Expand (3.5), subtract $\sum_j\mathbb E\|y_j\|^2$, and multiply by
$\theta/2$. \hfill$\square$

The right side of (3.3) is not nonnegative.  On one centered scalar root
$\xi$ with $\mathbb E\xi^2=1$, take
\[
 \rho_j^-=\xi,\qquad \rho_j^+=-\xi,\qquad \rho_*=0.
                                                               \tag{3.6}
\]
Then $\zeta_j=\xi$, $y_j=-\xi$, and $P_\ell\rho_*=0$.  At
$\theta=1/2$, both sides of (3.3) equal
\[
 -{1\over4}={1\over4}-{1\over2}.                      \tag{3.7}
\]
This is an algebraic method fixture, not a production counterexample.  It
proves that using $\theta\sum_j\|y_j\|^2/2$ is not the same operation as
removing $\theta\|\rho_*\|^2/2$ from the global terminal square.

The substitution is valid in the restrictive pure-future subcase
\[
 P_\ell\rho_*=0,\qquad \zeta_j=0\quad\hbox{for every }j, \tag{3.8}
\]
because then (3.3) vanishes.  The full terminal feedback does not satisfy
(3.8): its terminal coefficient and later selectors can depend on earlier
roots, and the moving prefix is generally nonzero.

% evidence-anchor: theorem-4.1-fractional-perspective
\subsection*{Theorem 4.1 (exact fractional perspective criterion)}

One has
\[
\boxed{
 \mathcal E_{\theta,R}
 ={1\over2}\|c+A_\theta^{-1}BG\|_{A_\theta}^2
 +{1\over2}G^T\Theta_{\theta,R}(B)G
 -{1\over2}B:\Gamma,}                                 \tag{4.3}
\]
and
\[
\boxed{
 \Theta_{\theta,R}(B)\succeq0
 \quad\Longleftrightarrow\quad
 2R\succeq\theta B.}                                  \tag{4.4}
\]

\emph{Proof.}
Expansion of the square in (4.3) proves the first identity.  Since
$A_\theta\succ0$, the Schur complement gives
\[
 \Theta_{\theta,R}(B)\succeq0
 \quad\Longleftrightarrow\quad
 \begin{pmatrix}B&B\\B&A_\theta\end{pmatrix}\succeq0.
                                                               \tag{4.5}
\]
The quadratic form of the block matrix is
\[
 (x+y)^TB(x+y)+y^T(2R-\theta B)y,                      \tag{4.6}
\]
so (4.5) is equivalent to $2R-\theta B\succeq0$.
\hfill$\square$

The failure below the threshold is already exact in one dimension.  Let
\[
 B=1,\qquad \theta={1\over2},\qquad R={1\over8},\qquad
 G=1,\qquad\Gamma=0.                                   \tag{4.7}
\]
Then
\[
 A_\theta={3\over4},\qquad
 \Theta_{\theta,R}=-{1\over3},\qquad
 c_*=-{4\over3},                                      \tag{4.8}
\]
and the optimized value of (4.2) is $-1/6$.  The threshold is exactly
$R=1/4$.

The new domination cannot be supplied by one fixed matrix in the production
problem.  On a pure-doublet ray,
\[

 \qquad B_R(te_1)=4c_1g(te_1)\otimes g(te_1),          \tag{4.9}
\]
and therefore
\[

 e_1\otimes e_1.                                      \tag{4.10}
\]
No fixed past payment $R$ dominates $\theta B_R$ for all fields.  Choosing

\[
\boxed{

 ={2\over\varrho}
 \{(r_S\cdot h)(z\cdot k)-(r_S\cdot k)(z\cdot h)\}.} \tag{7.2}

Since $r_S\wedge z=Sz\wedge z$ and $\|S\|_{\rm op}\le1$,

For $u_t=z+ta$, $G_i=D_iz$, and $c_i=D_ia$, the homotopy formula gives

\boxed{
\begin{aligned}

 -{5\over9}\int_0^1\Omega_{u_t}(a,G_i+tc_i)\,dt.
\end{aligned}}                                                \tag{7.4}

where
\[
