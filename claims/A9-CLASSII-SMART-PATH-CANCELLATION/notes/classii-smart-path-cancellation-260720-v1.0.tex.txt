\section*{1. Purpose and scope}

\begin{center}
\fbox{\parbox{0.91\textwidth}{
\textbf{What closes here.}  At every finite cutoff this note derives the
exact independent-to-self-coupled smart path, integrates the independent
carrier, and cancels all apparent trace-class terms by Gaussian integration
by parts.  It also proves an arbitrary-source, past-measurable frozen-shell
determinant/entropy theorem and its summable dyadic Schatten gain.

\medskip
\textbf{What remains open.}  The surviving commutator must still be bounded
uniformly under the \emph{tilted interacting laws}.  Base-Gaussian moments,
$L^2$ convergence of the A7 composite, and the A8 decoupled determinant do not
give that bound.  Consequently this note does not prove the self-coupled
Nelson estimate or the full three-component Gibbs measure.
}}
\end{center}

The model is the hash-pinned A1 production covariance on $\T_L^3$, $L=16$,
in the six-real convention.  The density floor is fixed at
$\varepsilon_\rho=10^{-12}$, $Q_{\rm II}$ is positive, products are exact or
sufficiently de-aliased, and every regulator is a common real-even scalar
multiplier from the A7 class.  Floor removal, asymmetric regulators, and
infinite volume are outside the theorem.

\section{Finite-cutoff setup}

Let $E_\Lambda$ be the finite white-noise space and write
\[
 K_\Lambda=R_\Lambda C^{1/2},\qquad
 G_\Lambda=DK_\Lambda,\qquad X=K_\Lambda x,
 \tag{2.1}
\]
where $x,y$ are independent standard Gaussians on $E_\Lambda$.  The A7/A8
coefficient is
\[
 B(z)=\sum_{A=1}^3
 \{a p_Ap_A^T+b(p_Av_A^T+v_Ap_A^T)+cv_Av_A^T\}\ge0,       \tag{2.2}
\]
with $p_A=2S_Az$, $v_A=2(S_A-q_A\Id)z$, and
$q_A=z^TS_Az/(|z|^2+\varepsilon_\rho)$.  Define
\[
 T(x)=G_\Lambda^*M_{B(K_\Lambda x)}G_\Lambda\ge0.          \tag{2.3}
\]
For $p>0$ and
\[
 z_s=sx+\sqrt{1-s^2}\,y,
\]
put
\[
 H_s(x,y)=U(K_\Lambda x)
 +{1\over2}\{z_s^TT(x)z_s-\Tr T(x)\},
 \qquad
 \Phi_\Lambda(s)=\log\E e^{-pH_s}.                        \tag{2.4}
\]
At $s=0$ this is exactly the A8 independent-carrier endpoint.  At $s=1$ it
is the finite-cutoff A7 self-coupled action.

All derivatives below are ordinary finite-dimensional derivatives.  Thus no
limiting interchange is hidden in the identities.  Write
$T_i=DT(x)[e_i]$.  Gaussian covariance differentiation gives, for
$0<s<1$,
\[
 \Phi_\Lambda'(s)
 =\E_{\nu_s}\left[p^2\langle H_x,H_z\rangle
                 -p\Tr H_{xz}\right],                    \tag{2.5}
\]
where $\nu_s$ is the tilted probability measure and
\[
 H_z=Tz,
 \quad
 (H_x)_i=U_i+{1\over2}(z^TT_i z-\Tr T_i),
 \quad
 \Tr H_{xz}=\sum_i(T_i z)_i.                              \tag{2.6}
\]
Consequently
\[
 \Phi_\Lambda'(s)=\E_{\nu_s}\left[
 p^2\langle DU,Tz\rangle
 +{p^2\over2}\sum_i(z^TT_i z-\Tr T_i)(Tz)_i
 -p\sum_i(T_i z)_i\right].                                \tag{2.7}
\]
Only $DB$ occurs in this first derivative.  Higher coefficient derivatives
appear only when the centered terms are estimated or integrated by parts.

\section{Conditional determinant and exact cancellation}

Set
\[
 t=1-s^2,\qquad R_t=(\Id+ptT)^{-1},\qquad S_t=TR_t.         \tag{3.1}
\]
Integrating $y$ in (2.4) exactly yields
\[
 \Phi_{\Lambda,t}=\log\E_x e^{W_t(x)},                    \tag{3.2}
\]
\[
 W_t=-pU-{1\over2}\log\det{}_2(\Id+ptT)
 +{p(1-t)\over2}\{\Tr T-x^TS_tx\}.                       \tag{3.3}
\]
Here $t=1$ is the A8 endpoint and $t=0$ is the A7 endpoint.  Direct
differentiation gives
\[
 \partial_tW_t={p\over2}(x^TS_tx-\Tr S_t)
 +{p^2(1-t)\over2}\|S_tx\|^2.                             \tag{3.4}
\]
This expression seems dangerous: neither $T$ nor $S_t$ is uniformly trace
class.  That appearance is false.

Let $a_t$ denote the coefficient-derivative part of $\nabla_xW_t$:
\[
 (a_t)_i={p^2t^2\over2}\Tr(TR_tT_i)
 +{p(1-t)\over2}\{\Tr T_i-x^TR_tT_iR_tx\}.                \tag{3.5}
\]
Since $DS_t[H]=R_tHR_t$,
\[
 \nabla_xW_t=-pDU-p(1-t)S_tx+a_t.                          \tag{3.6}
\]
Apply Gaussian integration by parts to the vector field $S_t(x)x$:
\[
 \E_{\nu_t}[x^TS_tx-\Tr S_t]
 =\E_{\nu_t}\left[
 \langle\diver S_t,x\rangle+\langle S_tx,\nabla_xW_t\rangle
 \right].                                                  \tag{3.7}
\]
Substitution of (3.6) into (3.7) cancels the last term in (3.4) exactly and
proves the central identity
\[
 \boxed{
 \partial_t\Phi_{\Lambda,t}
 ={p\over2}\E_{\nu_t}\left[
 -p\langle DU,S_tx\rangle
 +\langle a_t,S_tx\rangle
 +\langle\diver S_t,x\rangle\right].}                     \tag{3.8}
\]
No estimate was used in this cancellation.

\section{Why no Schatten-one term survives}

The trace difference in (3.5) has the exact decomposition
\[
 \Tr T_i-x^TR_tT_iR_tx
 =\Tr[(\Id-R_t^2)T_i]
 -\{x^TR_tT_iR_tx-\Tr(R_tT_iR_t)\}.                       \tag{4.1}
\]
Moreover
\[
 \Id-R_t^2=ptS_t(\Id+R_t).                                \tag{4.2}
\]
The first term in (4.1) is therefore an $\Schat$--$\Schat$ pairing, while
the second is an explicitly centered Hilbert--Schmidt quadratic form.  Thus
one must not estimate $\Tr T_i$ separately.

There is a second exact cancellation.  In coordinates,
\[
 T_{jk}(x)=\langle G_\Lambda e_j,
 B(K_\Lambda x)G_\Lambda e_k\rangle_{L^2}.                \tag{4.3}
\]
The contraction in $\sum_i\partial_iT_{ik}$ contains
\[
 \sum_i(G_\Lambda e_i)(r)(K_\Lambda e_i)(r)^T
 =D_rC_\Lambda(r,r).                                      \tag{4.4}
\]
Common real-even regulation makes (4.4) zero, hence
\[
 \diver T=0.                                               \tag{4.5}
\]
Using $R_t=\Id-ptS_t$ on the left index of $DS_t[e_i]$ gives
\[
 \diver S_t=-pt\operatorname{contr}_i(S_tT_iR_t),         \tag{4.6}
\]
and therefore
\[
 \|\diver S_t\|
 \le pt\|S_t\|_{\Schat}
 \left(\sum_i\|T_i\|_{\Schat}^2\right)^{1/2}.           \tag{4.7}
\]
An odd or component-split regulator need not satisfy (4.4).  The independent
verifier includes such a negative control and confirms that omitting
$\diver S_t$ then changes the derivative.

At fixed positive floor, $B\in C^\infty$ and the elementary bounds
\[
 \|B(z)\|_F\le\beta_B|z|^2,
 \qquad \|DB(z)\|\le\beta_1|z|                            \tag{4.8}
\]
hold with finite production constants.  The pinned values include
\[
 \beta_B=0.256499999999936.                                \tag{4.9}
\]
A8's spatial trace-ideal estimate applied to $DB(X)[K_\Lambda e_i]$ gives
\[
 \sum_i\|T_i\|_{\Schat}^2
 \le C_A\beta_1^2\kappa_0
 \int_{\T_L^3}(1+|X|)^2\dd r,                             \tag{4.10}
\]
where
\[
 \kappa_0=\sup_{\Lambda,r}\Tr C_\Lambda(r,r)<\infty       \tag{4.11}
\]
because the production covariance has order $-4$ in dimension three.
Analogous bounds for differentiated centered quadratics use $D^2B$ and
$\kappa_0^2$.  Thus every factor in (3.8) has cutoff-uniform moments under
the \emph{base} Gaussian law.  Section 7 explains why this is not yet the
needed tilted-law estimate.

\section{Arbitrary-source frozen-shell theorem}

The positive result needed for dyadic freezing is stronger than the A8
zero-source determinant.  Condition on all previously exposed shells.  Let
$b(r)\ge0$ and a derivative background $a(r)$ be arbitrary measurable
functions of that past.  For the new standard Gaussian shell $\xi_j$, put
\[
 A=M_b^{1/2}G_j,\qquad T=A^*A,\qquad q=M_b^{1/2}a,         \tag{5.1}
\]
\[
 Q={1\over2}\|q+A\xi_j\|_2^2-{1\over2}\Tr T.             \tag{5.2}
\]
Completing the finite-dimensional Gaussian square proves, for every $p>0$,
\[
\begin{split}
 \log\E_j e^{-pQ}
={}&{1\over2}\{p\Tr T-\log\det(\Id+pT)\}
 -{p\over2}\|q\|^2\\
 &+{p^2\over2}
 \langle A^*q,(\Id+pT)^{-1}A^*q\rangle.
\end{split}                                                \tag{5.3}
\]
The source contribution is nonpositive because
\[
 pA(\Id+pA^*A)^{-1}A^*\le\Id.                             \tag{5.4}
\]
For every eigenvalue $\lambda\ge0$,
$p\lambda-\log(1+p\lambda)\le p^2\lambda^2/2$.  Hence
\[
 \boxed{\log\E_j e^{-pQ}\le{p^2\over4}\|T\|_{\Schat}^2.}\tag{5.5}
\]
The Gibbs variational formula applied with $p=1/\eta$ gives the equivalent
entropy estimate
\[
 \E_{\nu_j}Q\ge-\eta H(\nu_j\mid\gamma_j)
 -{1\over4\eta}\|T\|_{\Schat}^2.                         \tag{5.6}
\]
Equations (5.3)--(5.6) retain an arbitrary low-shell derivative source.  This
is essential: discarding it would lose the positive energy needed to absorb
cross-shell terms.

\section{Dyadic gain and harmless independent shifts}

Let $P_j$ select a dyadic annulus and
$G_j=DP_jR_\Lambda C^{1/2}$.  The symbol lower bound
\[
 A(k)\ge c_{\rm sym}\langle k\rangle^4\Id,
 \qquad c_{\rm sym}=0.108364530587115                       \tag{6.1}
\]
implies that each differentiated covariance root is bounded by a constant
times $\langle k\rangle^{-1}$.  Both legs of $T_{j,b}$ lie in the same
annulus.  Uniformly in the transfer momentum,
\[
 \sum_{\substack{k,k-q\ \mathrm{in\ annulus}\ j}}
 \langle k\rangle^{-2}\langle k-q\rangle^{-2}
 \le C_{\rm sh}(L)2^{-j}.                                 \tag{6.2}
\]
Indeed the annulus has $O_L(2^{3j})$ modes and each product is
$O_L(2^{-4j})$; the finitely many low shells are absorbed into the constant.
Therefore
\[
 \|T_{j,b}\|_{\Schat}^2
 \le C_{\rm sh}(L)M_R^4c_{\rm sym}^{-2}
 2^{-j}\|b\|_{L^2}^2.                                    \tag{6.3}
\]
For $b=B(z)$, (4.8) turns (6.3) into a summable multiple of
$2^{-j}\|z\|_{L^4}^4$, which the positive sextic can absorb.

There is also an exact sign result.  If $h\in H^2$ is deterministic, or is
random but independent of $X_\Lambda$, same-point common evenness makes
$X_\Lambda(r)$ independent of $D_iX_\Lambda(r)$.  Conditional on $h$, the
$D_iX$ quadratic term cancels the exact A7 counterterm and the cross term has
zero mean.  Thus
\[
 \E_XV_\Lambda^{\rm ren}(X_\Lambda+h)
 ={1\over2}\sum_i\int
 \E_X[(D_ih)^TB(X_\Lambda+h)D_ih]\dd r\ge0.               \tag{6.4}
\]
The obstruction is therefore not an ordinary Cameron--Martin translation;
it is genuine same-noise adapted dependence.

\section{Exact remaining gate}

For one designated dyadic route, let $\phi_j=P_{\le j}\phi$ and
$\Delta_jB=B(\phi_j)-B(\phi_{j-1})$.  After applying Section 5 to the frozen
part, the unaccounted increment is exactly
\[
 \mathcal C_j={1\over2}\int(D\phi_j)^T\Delta_jB\,D\phi_j
 -{1\over2}\int\Tr[\Gamma_{\le j}\Delta_jB].             \tag{7.1}
\]
A sufficient closure estimate for this route is: for every $\eta>0$,
uniformly in $J$ and every tilted law $\nu$,
\[
 \E_\nu\sum_{j\le J}\mathcal C_j
 \ge-\eta H(\nu\mid\gamma_J)
 -\eta\E_\nu\|\phi_J\|_6^6-C_\eta.                       \tag{7.2}
\]
The corresponding smart-path target is a cutoff-uniform bound on the
negative variation of (3.8).  Either estimate would join the A8 endpoint to
the A7 endpoint and prove the desired Nelson bound.

Under the unshifted Gaussian law, power counting gives summable shell
increments and the leading cubic commutator has root-mean-square order
$2^{-j}$.  This is not (7.2): relative-entropy tilts may concentrate on rare
configurations, and an $L^2(\gamma)$ estimate does not control every such law.
The non-exact Class-II $K$ one-form also forbids replacing
$B(h)Dh$ by the derivative of a scalar cubic primitive.  Positive derivative
energy and the noncentral determinant cancellation must be retained.

The umbrella gate remains
\texttt{A7-CLASSII-SELF-COUPLING-INTERPOLATION}; its sharp residual after this
note is
\texttt{A7-CLASSII-TILTED-COMMUTATOR-FORM-BOUND}.

\section{Devil's-advocate review}

\begin{enumerate}
\item \textbf{``The raw $\Tr T$ term diverges, so the interpolation fails.''}
\textbf{DISMISSED.}  Equations (3.8), (4.1), and (4.2) remove it exactly;
only Hilbert--Schmidt pairings remain.

\item \textbf{``$\diver T=0$ for every regulator.''}
\textbf{UPHELD as false.}  It uses the common-even same-point identity
(4.4).  The executable odd control produces a nonzero correction.

\item \textbf{``The frozen-shell theorem only treats zero background.''}
\textbf{DISMISSED.}  Equation (5.3) includes arbitrary $q=M_b^{1/2}a$, and
(5.4) makes its net contribution nonpositive.

\item \textbf{``Base-law moments prove the tilted estimate.''}
\textbf{UPHELD as false.}  This is precisely the distinction between (4.10)
and (7.2).  The latter remains open.

\item \textbf{``Direct Malliavin integration by parts proves the adapted
drift bound.''}  \textbf{UPHELD as invalid.}  For an adapted shift $H$, the
identity contains $\Tr(Db\,D_xH)$, while the Boue--Dupuis cost controls $H$
but not its Malliavin derivative.

\item \textbf{``A new mass or vacuum counterterm has been ruled out at all
orders.''}  \textbf{UPHELD as an overclaim.}  No such term survives the first
exact interpolation reduction, but the all-scale tilted commutator audit is
unfinished and could still expose one.

\item \textbf{``Deterministic lower-bound counterexamples disprove this
route.''}  \textbf{DISMISSED.}  Equation (6.4) shows that deterministic and
independent translations are nonnegative after exact covariance centering.

\item \textbf{``The dyadic $2^{-j}$ factor is a fitted slope.''}
\textbf{DISMISSED.}  It is the analytic mode count (6.2).  Executable finite
matrices test signs and factors, not the all-shell proof.

\item \textbf{``This note constructs the physical Gibbs measure.''}
\textbf{UPHELD as false.}  The measure remains behind (7.2) or the equivalent
tilted smart-path variation bound.
\end{enumerate}
